@dataclass
class Goal:
    id: str
    description: str
    priority: int  # 1-5
    status: str  # 'active', 'completed', 'deferred'
    created_at: datetime
    completed_at: Optional[datetime] = None

@dataclass
class Decision:
    id: str
    description: str
    rationale: str
    made_at: datetime
    related_goals: List[str] = field(default_factory=list)

@dataclass
class Pattern:
    id: str
    name: str
    description: str
    frequency: int  # 발견 횟수
    first_seen: datetime
    last_seen: datetime

@dataclass
class Mistake:
    id: str
    description: str
    impact: str  # 'low', 'medium', 'high'
    resolution: Optional[str] = None
    occurred_at: datetime
    resolved_at: Optional[datetime] = None

@dataclass
class FileMapping:
    path: str
    purpose: str
    last_modified: datetime
    related_goals: List[str] = field(default_factory=list)

@dataclass
class Dependency:
    name: str
    version: Optional[str]
    purpose: str
    required: bool = True

@dataclass
class Risk:
    id: str
    description: str
    severity: str  # 'low', 'medium', 'high', 'critical'
    likelihood: str  # 'low', 'medium', 'high'
    mitigation: Optional[str] = None
    status: str = 'open'  # 'open', 'mitigated', 'accepted'

@dataclass
class LeaderContext:
    project_id: str
    phase: int
    goals: List[Goal] = field(default_factory=list)
    constraints: List[str] = field(default_factory=list)
    decisions: List[Decision] = field(default_factory=list)
    patterns: List[Pattern] = field(default_factory=list)
    mistakes: List[Mistake] = field(default_factory=list)
    file_map: List[FileMapping] = field(default_factory=list)
    dependencies: List[Dependency] = field(default_factory=list)
    risks: List[Risk] = field(default_factory=list)
    progress: Dict[str, Any] = field(default_factory=dict)
    domain: Dict[str, Any] = field(default_factory=dict)
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)
```

class ContextValidator:
    @staticmethod
    def validate(context: LeaderContext) -> List[str]:
        """컨텍스트 유효성 검사, 에러 목록 반환"""
        errors = []
        # project_id 필수
        # phase는 양수
        # goals의 priority는 1-5 범위
        # risks의 severity는 유효한 값
        return errors

    @staticmethod
    def is_valid(context: LeaderContext) -> bool:
        """유효성 검사 통과 여부"""
        return len(ContextValidator.validate(context)) == 0
```
