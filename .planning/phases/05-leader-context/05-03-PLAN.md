from typing import Optional, List
from dataclasses import dataclass
from datetime import datetime

@dataclass
class QueryFilter:
    status: Optional[str] = None
    priority_min: Optional[int] = None
    priority_max: Optional[int] = None
    severity: Optional[str] = None
    date_from: Optional[datetime] = None
    date_to: Optional[datetime] = None
    keywords: Optional[List[str]] = None

class ContextQuery:
    def __init__(self, context: LeaderContext):
        self.context = context

    # Goal 조회
    def get_active_goals(self) -> List[Goal]:
        """활성 목표 목록 반환"""
        return [g for g in self.context.goals if g.status == 'active']

    def get_goals_by_priority(self, min_priority: int = 1) -> List[Goal]:
        """우선순위 기준 목표 조회 (높은 순)"""
        return sorted(
            [g for g in self.context.goals if g.priority >= min_priority],
            key=lambda g: g.priority,
            reverse=True
        )

    def find_goals(self, keyword: str) -> List[Goal]:
        """키워드로 목표 검색"""
        keyword_lower = keyword.lower()
        return [g for g in self.context.goals
                if keyword_lower in g.description.lower()]

    # Decision 조회
    def get_recent_decisions(self, limit: int = 10) -> List[Decision]:
        """최근 결정사항 조회"""
        return sorted(
            self.context.decisions,
            key=lambda d: d.made_at,
            reverse=True
        )[:limit]

    def get_decisions_for_goal(self, goal_id: str) -> List[Decision]:
        """특정 목표 관련 결정사항"""
        return [d for d in self.context.decisions
                if goal_id in d.related_goals]

    def find_decisions(self, keyword: str) -> List[Decision]:
        """키워드로 결정사항 검색"""
        keyword_lower = keyword.lower()
        return [d for d in self.context.decisions
                if keyword_lower in d.description.lower()
                or keyword_lower in d.rationale.lower()]

    # Pattern 조회
    def get_frequent_patterns(self, min_frequency: int = 2) -> List[Pattern]:
        """자주 발견되는 패턴"""
        return sorted(
            [p for p in self.context.patterns if p.frequency >= min_frequency],
            key=lambda p: p.frequency,
            reverse=True
        )

    def get_recent_patterns(self, limit: int = 5) -> List[Pattern]:
        """최근 발견된 패턴"""
        return sorted(
            self.context.patterns,
            key=lambda p: p.last_seen,
            reverse=True
        )[:limit]

    # Mistake 조회
    def get_unresolved_mistakes(self) -> List[Mistake]:
        """미해결 실수 목록"""
        return [m for m in self.context.mistakes if m.resolved_at is None]

    def get_high_impact_mistakes(self) -> List[Mistake]:
        """높은 영향도 실수"""
        return [m for m in self.context.mistakes if m.impact == 'high']

    # FileMapping 조회
    def get_file_context(self, file_path: str) -> Optional[FileMapping]:
        """특정 파일의 컨텍스트"""
        for fm in self.context.file_map:
            if fm.path == file_path:
                return fm
        return None

    def get_files_for_goal(self, goal_id: str) -> List[FileMapping]:
        """특정 목표 관련 파일들"""
        return [fm for fm in self.context.file_map
                if goal_id in fm.related_goals]

    def find_files(self, keyword: str) -> List[FileMapping]:
        """키워드로 파일 검색"""
        keyword_lower = keyword.lower()
        return [fm for fm in self.context.file_map
                if keyword_lower in fm.path.lower()
                or keyword_lower in fm.purpose.lower()]

    # Risk 조회
    def get_open_risks(self) -> List[Risk]:
        """미해결 리스크"""
        return [r for r in self.context.risks if r.status == 'open']

    def get_critical_risks(self) -> List[Risk]:
        """치명적 리스크"""
        return [r for r in self.context.risks
                if r.severity == 'critical' and r.status == 'open']

    def get_risks_by_severity(self, severity: str) -> List[Risk]:
        """심각도별 리스크"""
        return [r for r in self.context.risks if r.severity == severity]

    # Dependency 조회
    def get_required_dependencies(self) -> List[Dependency]:
        """필수 의존성"""
        return [d for d in self.context.dependencies if d.required]

    def find_dependency(self, name: str) -> Optional[Dependency]:
        """이름으로 의존성 검색"""
        for d in self.context.dependencies:
            if d.name.lower() == name.lower():
                return d
        return None

    # 복합 조회
    def search_all(self, keyword: str) -> dict:
        """모든 항목에서 키워드 검색"""
        return {
            'goals': self.find_goals(keyword),
            'decisions': self.find_decisions(keyword),
            'files': self.find_files(keyword),
            'patterns': [p for p in self.context.patterns
                        if keyword.lower() in p.name.lower()
                        or keyword.lower() in p.description.lower()],
            'risks': [r for r in self.context.risks
                     if keyword.lower() in r.description.lower()]
        }

    def get_summary(self) -> dict:
        """컨텍스트 요약 정보"""
        return {
            'project_id': self.context.project_id,
            'phase': self.context.phase,
            'active_goals': len(self.get_active_goals()),
            'total_goals': len(self.context.goals),
            'decisions_count': len(self.context.decisions),
            'patterns_count': len(self.context.patterns),
            'unresolved_mistakes': len(self.get_unresolved_mistakes()),
            'files_tracked': len(self.context.file_map),
            'dependencies_count': len(self.context.dependencies),
            'open_risks': len(self.get_open_risks()),
            'critical_risks': len(self.get_critical_risks()),
            'updated_at': self.context.updated_at.isoformat()
        }

class LeaderContextService:
    """LeaderContext 통합 서비스"""

    def __init__(self, project_path: str):
        self.project_path = project_path
        self.storage = ContextStorage(project_path)
        self._context: Optional[LeaderContext] = None
        self._query: Optional[ContextQuery] = None

    @property
    def context(self) -> Optional[LeaderContext]:
        """현재 로드된 컨텍스트"""
        return self._context

    @property
    def query(self) -> Optional[ContextQuery]:
        """쿼리 인터페이스"""
        if self._context and not self._query:
            self._query = ContextQuery(self._context)
        return self._query

    def load(self, phase: Optional[int] = None) -> bool:
        """컨텍스트 로드"""
        self._context = self.storage.load(phase)
        self._query = None  # 쿼리 객체 리셋
        return self._context is not None

    def load_or_create(self, phase: int) -> LeaderContext:
        """로드하거나 새로 생성"""
        if not self.load(phase):
            self._context = ContextInitializer.create_empty(
                project_id=Path(self.project_path).name,
                phase=phase
            )
            self._query = None
        return self._context

    def save(self) -> bool:
        """현재 컨텍스트 저장"""
        if self._context:
            return self.storage.save(self._context)
        return False

    # 편의 메서드 - Goal
    def add_goal(self, description: str, priority: int = 3) -> Goal:
        """목표 추가"""
        goal = Goal(
            id=f"goal-{uuid.uuid4().hex[:8]}",
            description=description,
            priority=priority,
            status='active',
            created_at=datetime.now()
        )
        self._context.goals.append(goal)
        self._context.updated_at = datetime.now()
        return goal

    def complete_goal(self, goal_id: str) -> bool:
        """목표 완료 처리"""
        for goal in self._context.goals:
            if goal.id == goal_id:
                goal.status = 'completed'
                goal.completed_at = datetime.now()
                self._context.updated_at = datetime.now()
                return True
        return False

    # 편의 메서드 - Decision
    def add_decision(self, description: str, rationale: str,
                     related_goals: List[str] = None) -> Decision:
        """결정사항 추가"""
        decision = Decision(
            id=f"dec-{uuid.uuid4().hex[:8]}",
            description=description,
            rationale=rationale,
            made_at=datetime.now(),
            related_goals=related_goals or []
        )
        self._context.decisions.append(decision)
        self._context.updated_at = datetime.now()
        return decision

    # 편의 메서드 - Pattern
    def record_pattern(self, name: str, description: str) -> Pattern:
        """패턴 기록 (기존 있으면 frequency 증가)"""
        for pattern in self._context.patterns:
            if pattern.name == name:
                pattern.frequency += 1
                pattern.last_seen = datetime.now()
                self._context.updated_at = datetime.now()
                return pattern

        pattern = Pattern(
            id=f"pat-{uuid.uuid4().hex[:8]}",
            name=name,
            description=description,
            frequency=1,
            first_seen=datetime.now(),
            last_seen=datetime.now()
        )
        self._context.patterns.append(pattern)
        self._context.updated_at = datetime.now()
        return pattern

    # 편의 메서드 - Mistake
    def record_mistake(self, description: str, impact: str = 'medium') -> Mistake:
        """실수 기록"""
        mistake = Mistake(
            id=f"mis-{uuid.uuid4().hex[:8]}",
            description=description,
            impact=impact,
            occurred_at=datetime.now()
        )
        self._context.mistakes.append(mistake)
        self._context.updated_at = datetime.now()
        return mistake

    def resolve_mistake(self, mistake_id: str, resolution: str) -> bool:
        """실수 해결 기록"""
        for mistake in self._context.mistakes:
            if mistake.id == mistake_id:
                mistake.resolution = resolution
                mistake.resolved_at = datetime.now()
                self._context.updated_at = datetime.now()
                return True
        return False

    # 편의 메서드 - Risk
    def add_risk(self, description: str, severity: str = 'medium',
                 likelihood: str = 'medium', mitigation: str = None) -> Risk:
        """리스크 추가"""
        risk = Risk(
            id=f"risk-{uuid.uuid4().hex[:8]}",
            description=description,
            severity=severity,
            likelihood=likelihood,
            mitigation=mitigation,
            status='open'
        )
        self._context.risks.append(risk)
        self._context.updated_at = datetime.now()
        return risk

    def mitigate_risk(self, risk_id: str, mitigation: str) -> bool:
        """리스크 완화 기록"""
        for risk in self._context.risks:
            if risk.id == risk_id:
                risk.mitigation = mitigation
                risk.status = 'mitigated'
                self._context.updated_at = datetime.now()
                return True
        return False

    # 편의 메서드 - FileMapping
    def track_file(self, path: str, purpose: str,
                   related_goals: List[str] = None) -> FileMapping:
        """파일 추적"""
        # 기존 파일이면 업데이트
        for fm in self._context.file_map:
            if fm.path == path:
                fm.purpose = purpose
                fm.last_modified = datetime.now()
                if related_goals:
                    fm.related_goals = list(set(fm.related_goals + related_goals))
                self._context.updated_at = datetime.now()
                return fm

        fm = FileMapping(
            path=path,
            purpose=purpose,
            last_modified=datetime.now(),
            related_goals=related_goals or []
        )
        self._context.file_map.append(fm)
        self._context.updated_at = datetime.now()
        return fm

    # 진행률 업데이트
    def update_progress(self, key: str, value: any) -> None:
        """진행 상황 업데이트"""
        self._context.progress[key] = value
        self._context.updated_at = datetime.now()

    # 도메인 지식 업데이트
    def update_domain(self, key: str, value: any) -> None:
        """도메인 지식 업데이트"""
        self._context.domain[key] = value
        self._context.updated_at = datetime.now()
