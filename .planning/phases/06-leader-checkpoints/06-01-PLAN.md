```python
from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List, Dict, Any

class CheckpointTrigger(Enum):
    """체크포인트 트리거 유형"""
    SUBTASK_COMPLETE = "subtask_complete"  # 서브태스크 완료
    ERROR_OCCURRED = "error_occurred"      # 에러 발생
    DECISION_MADE = "decision_made"        # 주요 결정
    PERIODIC = "periodic"                   # 주기적 (N개 작업마다)
    EXPLICIT = "explicit"                   # 명시적 요청
    PHASE_START = "phase_start"            # Phase 시작
    PHASE_END = "phase_end"                # Phase 종료

@dataclass
class CheckpointConfig:
    """체크포인트 설정"""
    enabled: bool = True
    periodic_interval: int = 5              # N개 작업마다 체크포인트
    on_subtask_complete: bool = True
    on_error: bool = True
    on_decision: bool = True
    max_checkpoints: int = 50               # 최대 보관 개수
    auto_cleanup: bool = True               # 오래된 체크포인트 자동 삭제

@dataclass
class Checkpoint:
    """체크포인트 데이터"""
    id: str
    phase: int
    trigger: CheckpointTrigger
    timestamp: datetime
    context_snapshot: Dict[str, Any]        # LeaderContext 스냅샷
    trigger_details: Dict[str, Any] = field(default_factory=dict)
    # 예: subtask_id, error_message, decision_id 등

    # 메타데이터
    subtask_count: int = 0                  # 누적 서브태스크 수
    error_count: int = 0                    # 누적 에러 수
    decision_count: int = 0                 # 누적 결정 수
```

```python
class CheckpointManager:
    """체크포인트 생성 및 관리"""

    def __init__(self, project_path: str, config: CheckpointConfig = None):
        self.project_path = project_path
        self.config = config or CheckpointConfig()
        self.checkpoint_dir = Path(project_path) / ".planning" / "leader_context" / "checkpoints"
        self.context_service = LeaderContextService(project_path)

        # 카운터
        self._subtask_count = 0
        self._error_count = 0
        self._decision_count = 0
        self._periodic_counter = 0

    def should_create_checkpoint(self, trigger: CheckpointTrigger) -> bool:
        """체크포인트 생성 여부 판단"""
        if not self.config.enabled:
            return False

        if trigger == CheckpointTrigger.PERIODIC:
            return self._periodic_counter >= self.config.periodic_interval
        elif trigger == CheckpointTrigger.SUBTASK_COMPLETE:
            return self.config.on_subtask_complete
        elif trigger == CheckpointTrigger.ERROR_OCCURRED:
            return self.config.on_error
        elif trigger == CheckpointTrigger.DECISION_MADE:
            return self.config.on_decision
        else:
            return True  # EXPLICIT, PHASE_START, PHASE_END는 항상 생성

    def create_checkpoint(
        self,
        trigger: CheckpointTrigger,
        details: Dict[str, Any] = None
    ) -> Optional[Checkpoint]:
        """체크포인트 생성"""
        if not self.should_create_checkpoint(trigger):
            return None

        # 컨텍스트 스냅샷
        context = self.context_service.context
        if context is None:
            self.context_service.load()
            context = self.context_service.context

        checkpoint = Checkpoint(
            id=f"cp-{uuid.uuid4().hex[:8]}",
            phase=context.phase if context else 0,
            trigger=trigger,
            timestamp=datetime.now(),
            context_snapshot=LeaderContextSerializer.to_dict(context) if context else {},
            trigger_details=details or {},
            subtask_count=self._subtask_count,
            error_count=self._error_count,
            decision_count=self._decision_count
        )

        self._save_checkpoint(checkpoint)
        self._reset_periodic_counter()

        if self.config.auto_cleanup:
            self._cleanup_old_checkpoints()

        return checkpoint

    def on_subtask_complete(self, subtask_id: str, result: str = "success") -> Optional[Checkpoint]:
        """서브태스크 완료 시 호출"""
        self._subtask_count += 1
        self._periodic_counter += 1

        return self.create_checkpoint(
            CheckpointTrigger.SUBTASK_COMPLETE,
            {"subtask_id": subtask_id, "result": result}
        )

    def on_error(self, error_message: str, error_type: str = "unknown") -> Optional[Checkpoint]:
        """에러 발생 시 호출"""
        self._error_count += 1

        return self.create_checkpoint(
            CheckpointTrigger.ERROR_OCCURRED,
            {"error_message": error_message, "error_type": error_type}
        )

    def on_decision(self, decision_id: str, description: str) -> Optional[Checkpoint]:
        """주요 결정 시 호출"""
        self._decision_count += 1

        return self.create_checkpoint(
            CheckpointTrigger.DECISION_MADE,
            {"decision_id": decision_id, "description": description}
        )

    def check_periodic(self) -> Optional[Checkpoint]:
        """주기적 체크포인트 확인"""
        if self._periodic_counter >= self.config.periodic_interval:
            return self.create_checkpoint(CheckpointTrigger.PERIODIC)
        return None

    def list_checkpoints(self, phase: int = None) -> List[Checkpoint]:
        """체크포인트 목록 조회"""
        # 파일 시스템에서 로드

    def get_checkpoint(self, checkpoint_id: str) -> Optional[Checkpoint]:
        """특정 체크포인트 조회"""

    def restore_from_checkpoint(self, checkpoint_id: str) -> bool:
        """체크포인트에서 컨텍스트 복원"""

    def _save_checkpoint(self, checkpoint: Checkpoint) -> None:
        """체크포인트 저장"""

    def _cleanup_old_checkpoints(self) -> None:
        """오래된 체크포인트 정리"""

    def _reset_periodic_counter(self) -> None:
        """주기 카운터 리셋"""
        self._periodic_counter = 0
```

```python
class CheckpointSerializer:
    """체크포인트 JSON 직렬화"""

    @staticmethod
    def to_dict(checkpoint: Checkpoint) -> dict:
        """Checkpoint를 dictionary로 변환"""
        return {
            "id": checkpoint.id,
            "phase": checkpoint.phase,
            "trigger": checkpoint.trigger.value,
            "timestamp": checkpoint.timestamp.isoformat(),
            "context_snapshot": checkpoint.context_snapshot,
            "trigger_details": checkpoint.trigger_details,
            "subtask_count": checkpoint.subtask_count,
            "error_count": checkpoint.error_count,
            "decision_count": checkpoint.decision_count
        }

    @staticmethod
    def from_dict(data: dict) -> Checkpoint:
        """dictionary에서 Checkpoint 생성"""
        return Checkpoint(
            id=data["id"],
            phase=data["phase"],
            trigger=CheckpointTrigger(data["trigger"]),
            timestamp=datetime.fromisoformat(data["timestamp"]),
            context_snapshot=data["context_snapshot"],
            trigger_details=data.get("trigger_details", {}),
            subtask_count=data.get("subtask_count", 0),
            error_count=data.get("error_count", 0),
            decision_count=data.get("decision_count", 0)
        )

    @staticmethod
    def to_json(checkpoint: Checkpoint) -> str:
        """JSON 문자열로 직렬화"""
        return json.dumps(CheckpointSerializer.to_dict(checkpoint), indent=2)

    @staticmethod
    def from_json(json_str: str) -> Checkpoint:
        """JSON 문자열에서 역직렬화"""
        return CheckpointSerializer.from_dict(json.loads(json_str))
```
