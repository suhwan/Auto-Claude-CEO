```python
from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List, Dict, Any

class ReviewType(Enum):
    """리뷰 유형"""
    SUBTASK = "subtask"           # 서브태스크 완료 리뷰
    PHASE = "phase"               # Phase 완료 요약
    ERROR_RESOLUTION = "error"    # 에러 해결 교훈
    DECISION = "decision"         # 결정 회고
    MANUAL = "manual"             # 수동 리뷰

class ReviewRating(Enum):
    """리뷰 평가"""
    EXCELLENT = "excellent"       # 매우 좋음
    GOOD = "good"                 # 좋음
    ACCEPTABLE = "acceptable"     # 수용 가능
    NEEDS_IMPROVEMENT = "needs_improvement"  # 개선 필요
    POOR = "poor"                 # 부족

@dataclass
class ReviewCriteria:
    """리뷰 기준별 평가"""
    code_quality: Optional[ReviewRating] = None
    decision_quality: Optional[ReviewRating] = None
    efficiency: Optional[ReviewRating] = None
    reusability: Optional[ReviewRating] = None
    risk_management: Optional[ReviewRating] = None
    notes: Dict[str, str] = field(default_factory=dict)

@dataclass
class LearningPoint:
    """학습 포인트"""
    id: str
    type: str  # 'pattern', 'mistake', 'insight'
    description: str
    source_review_id: str
    confidence: float = 0.8  # 0-1, 적용 확신도
    applied: bool = False    # LeaderContext 반영 여부

@dataclass
class Review:
    """리뷰 항목"""
    id: str
    phase: int
    review_type: ReviewType
    timestamp: datetime
    subject: str              # 리뷰 대상 (subtask_id, phase_id 등)
    subject_description: str

    # 평가
    criteria: ReviewCriteria = field(default_factory=ReviewCriteria)
    overall_rating: Optional[ReviewRating] = None
    summary: str = ""

    # 학습 포인트
    learning_points: List[LearningPoint] = field(default_factory=list)

    # 개선 제안
    improvements: List[str] = field(default_factory=list)
    action_items: List[str] = field(default_factory=list)

    # 참조
    checkpoint_id: Optional[str] = None
    error_ids: List[str] = field(default_factory=list)
    decision_ids: List[str] = field(default_factory=list)

    # 메타
    auto_generated: bool = True
    reviewed_by: str = "system"  # system, user, agent
```

```python
class ReviewGenerator:
    """자동 리뷰 생성"""

    def __init__(
        self,
        project_path: str,
        checkpoint_manager: CheckpointManager,
        error_logger: ErrorLogger,
        context_service: LeaderContextService
    ):
        self.project_path = project_path
        self.checkpoint_manager = checkpoint_manager
        self.error_logger = error_logger
        self.context_service = context_service
        self.reviews_dir = Path(project_path) / ".planning" / "leader_context" / "reviews"
        self.reviews_dir.mkdir(parents=True, exist_ok=True)

        self._reviews: List[Review] = []
        self._load_reviews()

    def review_subtask(
        self,
        subtask_id: str,
        description: str,
        result: Dict[str, Any],
        checkpoint_id: str = None
    ) -> Review:
        """서브태스크 완료 리뷰"""
        review = Review(
            id=f"rev-{uuid.uuid4().hex[:8]}",
            phase=self.context_service.context.phase if self.context_service.context else 0,
            review_type=ReviewType.SUBTASK,
            timestamp=datetime.now(),
            subject=subtask_id,
            subject_description=description,
            checkpoint_id=checkpoint_id
        )

        # 자동 평가
        review.criteria = self._evaluate_subtask(result)
        review.overall_rating = self._calculate_overall_rating(review.criteria)
        review.summary = self._generate_summary(review)

        # 학습 포인트 추출
        review.learning_points = self._extract_learning_points(result)

        # 개선 제안
        review.improvements = self._suggest_improvements(review)

        self._reviews.append(review)
        self._apply_learning_points(review)
        self._save_reviews()

        return review

    def review_phase(self, phase: int) -> Review:
        """Phase 완료 요약 리뷰"""
        # Phase 내 모든 리뷰 집계
        phase_reviews = [r for r in self._reviews if r.phase == phase]
        errors = self.error_logger.get_error_summary()
        checkpoints = self.checkpoint_manager.list_checkpoints(phase)

        review = Review(
            id=f"rev-phase-{phase}",
            phase=phase,
            review_type=ReviewType.PHASE,
            timestamp=datetime.now(),
            subject=f"phase-{phase}",
            subject_description=f"Phase {phase} 완료 요약"
        )

        # 집계된 평가
        review.criteria = self._aggregate_criteria(phase_reviews)
        review.overall_rating = self._calculate_overall_rating(review.criteria)
        review.summary = self._generate_phase_summary(phase, phase_reviews, errors)

        # 집계된 학습 포인트
        all_learning = []
        for r in phase_reviews:
            all_learning.extend(r.learning_points)
        review.learning_points = self._deduplicate_learning_points(all_learning)

        # 개선 제안
        review.improvements = self._suggest_phase_improvements(phase_reviews, errors)
        review.action_items = self._generate_action_items(review)

        self._reviews.append(review)
        self._save_reviews()

        return review

    def review_error_resolution(
        self,
        error_id: str,
        resolution: str
    ) -> Review:
        """에러 해결 교훈 리뷰"""
        error = self.error_logger._find_by_id(error_id)
        if not error:
            return None

        review = Review(
            id=f"rev-err-{error_id}",
            phase=error.phase,
            review_type=ReviewType.ERROR_RESOLUTION,
            timestamp=datetime.now(),
            subject=error_id,
            subject_description=f"에러 해결: {error.message[:50]}",
            error_ids=[error_id]
        )

        # 교훈 추출
        learning = LearningPoint(
            id=f"lp-{uuid.uuid4().hex[:8]}",
            type="mistake",
            description=f"Error: {error.message}\nResolution: {resolution}",
            source_review_id=review.id
        )
        review.learning_points = [learning]

        # 미래 방지 제안
        review.improvements = self._suggest_prevention(error, resolution)

        self._reviews.append(review)
        self._apply_learning_points(review)
        self._save_reviews()

        return review

    def get_reviews(self, phase: int = None, review_type: ReviewType = None) -> List[Review]:
        """리뷰 조회"""
        result = self._reviews
        if phase is not None:
            result = [r for r in result if r.phase == phase]
        if review_type is not None:
            result = [r for r in result if r.review_type == review_type]
        return result

    def get_all_learning_points(self, applied_only: bool = False) -> List[LearningPoint]:
        """모든 학습 포인트 조회"""
        all_points = []
        for review in self._reviews:

    def get_improvement_summary(self) -> Dict[str, Any]:

        return {

            "learning_points": len(self.get_all_learning_points()),
            "applied_patterns": sum(

                if lp.type == "pattern" and lp.applied
            ),
            "recorded_mistakes": sum(

                if lp.type == "mistake" and lp.applied
            ),
            "pending_action_items": self._count_pending_actions()
```

```python
class LeaderCheckpointService:
    """Leader Checkpoint 통합 서비스"""

    def __init__(self, project_path: str, phase: int, config: CheckpointConfig = None):
        self.project_path = project_path
        self.phase = phase

        # 컴포넌트 초기화
        self.context_service = LeaderContextService(project_path)
        self.context_service.load_or_create(phase)

        self.checkpoint_manager = CheckpointManager(project_path, config)
        self.error_logger = ErrorLogger(project_path, phase)
        self.review_generator = ReviewGenerator(
            project_path,
            self.checkpoint_manager,
            self.error_logger,
            self.context_service
        )

    # 통합 API
    def start_phase(self) -> Checkpoint:
        """Phase 시작"""
        return self.checkpoint_manager.create_checkpoint(
            CheckpointTrigger.PHASE_START,
            {"phase": self.phase}
        )

    def complete_subtask(
        self,
        subtask_id: str,
        description: str,
        result: Dict[str, Any]
    ) -> Dict[str, Any]:
        """서브태스크 완료 처리"""
        # 1. 체크포인트 생성
        checkpoint = self.checkpoint_manager.on_subtask_complete(subtask_id, "success")

        # 2. 리뷰 생성
        review = self.review_generator.review_subtask(
            subtask_id, description, result,
            checkpoint.id if checkpoint else None
        )

        return {
            "checkpoint": checkpoint,
            "review": review
        }

    def log_error(
        self,
        message: str,
        severity: ErrorSeverity = ErrorSeverity.ERROR,
        category: ErrorCategory = ErrorCategory.UNKNOWN,
        stack_trace: str = None,
        context: ErrorContext = None
    ) -> Dict[str, Any]:
        """에러 로깅 및 체크포인트"""
        # 1. 체크포인트 생성
        checkpoint = self.checkpoint_manager.on_error(message, category.value)

        # 2. 에러 로깅
        error = self.error_logger.log_error(
            message, severity, category, stack_trace, context,
            checkpoint.id if checkpoint else None
        )

        return {
            "checkpoint": checkpoint,
            "error": error
        }

    def resolve_error(self, error_id: str, resolution: str) -> Dict[str, Any]:
        """에러 해결 및 리뷰"""
        # 1. 에러 해결 기록
        self.error_logger.resolve_error(error_id, resolution)

        # 2. 교훈 리뷰 생성
        review = self.review_generator.review_error_resolution(error_id, resolution)

        return {"review": review}

    def record_decision(self, description: str, rationale: str) -> Dict[str, Any]:
        """결정 기록 및 체크포인트"""
        # 1. LeaderContext에 결정 추가
        decision = self.context_service.add_decision(description, rationale)

        # 2. 체크포인트 생성
        checkpoint = self.checkpoint_manager.on_decision(decision.id, description)

        return {
            "decision": decision,
            "checkpoint": checkpoint
        }

    def end_phase(self) -> Dict[str, Any]:
        """Phase 종료"""
        # 1. 종료 체크포인트
        checkpoint = self.checkpoint_manager.create_checkpoint(
            CheckpointTrigger.PHASE_END,
            {"phase": self.phase}
        )

        # 2. Phase 리뷰 생성
        review = self.review_generator.review_phase(self.phase)

        # 3. 컨텍스트 저장
        self.context_service.save()

        return {
            "checkpoint": checkpoint,
            "review": review,
            "error_summary": self.error_logger.get_error_summary(),
            "improvement_summary": self.review_generator.get_improvement_summary()
        }

    def get_status(self) -> Dict[str, Any]:
        """현재 상태 조회"""
        return {
            "phase": self.phase,
            "checkpoints": len(self.checkpoint_manager.list_checkpoints(self.phase)),
            "errors": self.error_logger.get_error_summary(),
            "reviews": len(self.review_generator.get_reviews(self.phase)),
            "learning_points": len(self.review_generator.get_all_learning_points())
        }

    def restore_from_checkpoint(self, checkpoint_id: str) -> bool:
        """체크포인트에서 복원"""
        return self.checkpoint_manager.restore_from_checkpoint(checkpoint_id)
```
