from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime, timedelta
from typing import Optional, List, Dict, Any

class MeetingType(Enum):
    """미팅 유형"""
    STANDUP = "standup"          # 일일 상태 공유
    SYNC = "sync"                # 정기 동기화
    ESCALATION = "escalation"    # 블로커 해결
    REVIEW = "review"            # Phase 리뷰
    KICKOFF = "kickoff"          # Phase 시작
    RETROSPECTIVE = "retro"      # 회고

class MeetingStatus(Enum):
    """미팅 상태"""
    SCHEDULED = "scheduled"
    IN_PROGRESS = "in_progress"
    COMPLETED = "completed"
    CANCELLED = "cancelled"

@dataclass
class AgendaItem:
    """아젠다 항목"""
    id: str
    title: str
    description: str = ""
    presenter: Optional[str] = None
    time_minutes: int = 5
    status: str = "pending"  # pending, discussed, skipped
    notes: str = ""
    decisions: List[str] = field(default_factory=list)

@dataclass
class ActionItem:
    """액션 아이템"""
    id: str
    title: str
    assignee: str
    due_date: Optional[datetime] = None
    status: str = "pending"  # pending, in_progress, completed
    notes: str = ""
    created_at: datetime = field(default_factory=datetime.now)
    completed_at: Optional[datetime] = None

@dataclass
class MeetingParticipant:
    """미팅 참석자"""
    team_id: str
    team_name: str
    role: str = "attendee"  # facilitator, attendee, optional
    attended: bool = False
    joined_at: Optional[datetime] = None

@dataclass
class MeetingMinutes:
    """미팅 의사록"""
    summary: str = ""
    key_decisions: List[str] = field(default_factory=list)
    concerns_raised: List[str] = field(default_factory=list)
    next_steps: List[str] = field(default_factory=list)
    recorded_by: Optional[str] = None

@dataclass
class Meeting:
    """미팅"""
    id: str
    title: str
    meeting_type: MeetingType
    status: MeetingStatus = MeetingStatus.SCHEDULED

    # 일정
    scheduled_at: Optional[datetime] = None
    started_at: Optional[datetime] = None
    ended_at: Optional[datetime] = None
    duration_minutes: int = 30

    # 참석자
    participants: List[MeetingParticipant] = field(default_factory=list)
    facilitator: Optional[str] = None

    # 아젠다
    agenda: List[AgendaItem] = field(default_factory=list)
    auto_generated_agenda: bool = True

    # 결과
    action_items: List[ActionItem] = field(default_factory=list)
    minutes: MeetingMinutes = field(default_factory=MeetingMinutes)

    # 컨텍스트
    phase: int = 0
    related_sync_point_id: Optional[str] = None
    related_blocker_ids: List[str] = field(default_factory=list)

    # 메타데이터
    created_at: datetime = field(default_factory=datetime.now)
    metadata: Dict[str, Any] = field(default_factory=dict)

@dataclass
class MeetingSchedule:
    """미팅 스케줄"""
    id: str
    meeting_type: MeetingType
    recurrence: str = "none"  # none, daily, weekly, phase_start, phase_end
    teams: List[str] = field(default_factory=list)
    default_duration_minutes: int = 30
    default_agenda_template: List[str] = field(default_factory=list)
    enabled: bool = True
    next_occurrence: Optional[datetime] = None

class MeetingManager:
    """Leader Meeting 관리"""

    def __init__(
        self,
        project_path: str,
        sync_manager: SyncPointManager = None,
        board_manager: SharedBoardManager = None
    ):
        self.project_path = project_path
        self.sync_manager = sync_manager or SyncPointManager(project_path)
        self.board_manager = board_manager or SharedBoardManager(project_path, self.sync_manager)
        self.meetings_dir = Path(project_path) / ".planning" / "team_sync" / "meetings"
        self.meetings_dir.mkdir(parents=True, exist_ok=True)

        self._meetings: Dict[str, Meeting] = {}
        self._schedules: Dict[str, MeetingSchedule] = {}
        self._load()

    # 미팅 생성
    def create_meeting(
        self,
        title: str,
        meeting_type: MeetingType,
        teams: List[str],
        phase: int = 0,
        duration_minutes: int = 30,
        scheduled_at: datetime = None
    ) -> Meeting:
        """미팅 생성"""
        meeting = Meeting(
            id=f"meeting-{uuid.uuid4().hex[:8]}",
            title=title,
            meeting_type=meeting_type,
            phase=phase,
            duration_minutes=duration_minutes,
            scheduled_at=scheduled_at or datetime.now(),
            participants=[
                MeetingParticipant(team_id=t, team_name=t)
                for t in teams
            ]
        )

        # 자동 아젠다 생성
        meeting.agenda = self._generate_agenda(meeting_type, phase, teams)

        self._meetings[meeting.id] = meeting
        self._save()
        return meeting

    def create_escalation_meeting(
        self,
        blocker_ids: List[str],
        teams: List[str],
        phase: int = 0
    ) -> Meeting:
        """블로커 해결 미팅 생성"""
        meeting = self.create_meeting(
            title="Blocker Resolution Meeting",
            meeting_type=MeetingType.ESCALATION,
            teams=teams,
            phase=phase,
            duration_minutes=45
        )
        meeting.related_blocker_ids = blocker_ids

        # 블로커별 아젠다 항목 추가
        for blocker_id in blocker_ids:
            meeting.agenda.append(AgendaItem(
                id=f"agenda-{blocker_id}",
                title=f"Blocker: {blocker_id}",
                time_minutes=10
            ))

        self._save()
        return meeting

    def create_phase_kickoff(self, phase: int, teams: List[str]) -> Meeting:
        """Phase 시작 미팅"""
        meeting = self.create_meeting(
            title=f"Phase {phase} Kickoff",
            meeting_type=MeetingType.KICKOFF,
            teams=teams,
            phase=phase,
            duration_minutes=60
        )

        meeting.agenda = [
            AgendaItem(id="1", title="Phase 목표 공유", time_minutes=15),
            AgendaItem(id="2", title="팀별 계획 발표", time_minutes=20),
            AgendaItem(id="3", title="의존성 및 리스크 논의", time_minutes=15),
            AgendaItem(id="4", title="Sync Points 합의", time_minutes=10),
        ]

        self._save()
        return meeting

    def create_phase_review(self, phase: int, teams: List[str]) -> Meeting:
        """Phase 리뷰 미팅"""
        meeting = self.create_meeting(
            title=f"Phase {phase} Review",
            meeting_type=MeetingType.REVIEW,
            teams=teams,
            phase=phase,
            duration_minutes=60
        )

        meeting.agenda = [
            AgendaItem(id="1", title="목표 달성 현황", time_minutes=15),
            AgendaItem(id="2", title="팀별 성과 공유", time_minutes=20),
            AgendaItem(id="3", title="배운 점/개선점", time_minutes=15),
            AgendaItem(id="4", title="다음 Phase 준비", time_minutes=10),
        ]

        self._save()
        return meeting

    # 미팅 진행
    def start_meeting(self, meeting_id: str) -> Dict[str, Any]:
        """미팅 시작"""
        meeting = self._meetings.get(meeting_id)
        if not meeting:
            return {"success": False, "error": "Meeting not found"}

        meeting.status = MeetingStatus.IN_PROGRESS
        meeting.started_at = datetime.now()
        self._save()

        return {
            "success": True,
            "meeting": meeting,
            "agenda": meeting.agenda
        }

    def record_attendance(self, meeting_id: str, team_id: str) -> bool:
        """참석 기록"""
        meeting = self._meetings.get(meeting_id)
        if not meeting:
            return False

        for participant in meeting.participants:
            if participant.team_id == team_id:
                participant.attended = True
                participant.joined_at = datetime.now()
                self._save()
                return True
        return False

    def update_agenda_item(
        self,
        meeting_id: str,
        agenda_item_id: str,
        status: str,
        notes: str = "",
        decisions: List[str] = None
    ) -> bool:
        """아젠다 항목 업데이트"""
        meeting = self._meetings.get(meeting_id)
        if not meeting:
            return False

        for item in meeting.agenda:
            if item.id == agenda_item_id:
                item.status = status
                item.notes = notes
                if decisions:
                    item.decisions = decisions
                self._save()
                return True
        return False

    def add_action_item(
        self,
        meeting_id: str,
        title: str,
        assignee: str,
        due_date: datetime = None
    ) -> Optional[ActionItem]:
        """액션 아이템 추가"""
        meeting = self._meetings.get(meeting_id)
        if not meeting:
            return None

        action = ActionItem(
            id=f"action-{uuid.uuid4().hex[:8]}",
            title=title,
            assignee=assignee,
            due_date=due_date
        )
        meeting.action_items.append(action)
        self._save()
        return action

    def end_meeting(
        self,
        meeting_id: str,
        summary: str = "",
        key_decisions: List[str] = None,
        next_steps: List[str] = None
    ) -> Dict[str, Any]:
        """미팅 종료"""
        meeting = self._meetings.get(meeting_id)
        if not meeting:
            return {"success": False, "error": "Meeting not found"}

        meeting.status = MeetingStatus.COMPLETED
        meeting.ended_at = datetime.now()

        # 의사록 업데이트
        meeting.minutes.summary = summary
        meeting.minutes.key_decisions = key_decisions or []
        meeting.minutes.next_steps = next_steps or []

        # 자동 요약
        if not summary:
            meeting.minutes.summary = self._generate_summary(meeting)

        self._save()

        return {
            "success": True,
            "duration_actual": (meeting.ended_at - meeting.started_at).seconds // 60,
            "decisions_count": len(meeting.minutes.key_decisions),
            "action_items_count": len(meeting.action_items)
        }

    # 액션 아이템 관리
    def update_action_item(
        self,
        meeting_id: str,
        action_id: str,
        status: str,
        notes: str = ""
    ) -> bool:
        """액션 아이템 상태 업데이트"""
        meeting = self._meetings.get(meeting_id)
        if not meeting:
            return False

        for action in meeting.action_items:
            if action.id == action_id:
                action.status = status
                action.notes = notes
                if status == "completed":
                    action.completed_at = datetime.now()
                self._save()
                return True
        return False

    def get_pending_actions(self, team_id: str = None) -> List[Dict[str, Any]]:
        """미완료 액션 아이템 조회"""
        actions = []
        for meeting in self._meetings.values():
            for action in meeting.action_items:
                if action.status != "completed":
                    if team_id is None or action.assignee == team_id:
                        actions.append({
                            "meeting_id": meeting.id,
                            "meeting_title": meeting.title,
                            "action": action,
                            "overdue": action.due_date and action.due_date < datetime.now()
                        })
        return sorted(actions, key=lambda x: x["action"].due_date or datetime.max)

    # 스케줄 관리
    def create_schedule(
        self,
        meeting_type: MeetingType,
        teams: List[str],
        recurrence: str = "weekly",
        duration_minutes: int = 30
    ) -> MeetingSchedule:
        """정기 미팅 스케줄 생성"""
        schedule = MeetingSchedule(
            id=f"sched-{uuid.uuid4().hex[:8]}",
            meeting_type=meeting_type,
            recurrence=recurrence,
            teams=teams,
            default_duration_minutes=duration_minutes,
            next_occurrence=self._calculate_next_occurrence(recurrence)
        )
        self._schedules[schedule.id] = schedule
        self._save()
        return schedule

    def check_and_create_scheduled_meetings(self) -> List[Meeting]:
        """스케줄에 따른 미팅 자동 생성"""
        created = []
        now = datetime.now()

        for schedule in self._schedules.values():
            if not schedule.enabled:
                continue

            if schedule.next_occurrence and schedule.next_occurrence <= now:
                meeting = self.create_meeting(
                    title=f"{schedule.meeting_type.value.title()} Meeting",
                    meeting_type=schedule.meeting_type,
                    teams=schedule.teams,
                    duration_minutes=schedule.default_duration_minutes,
                    scheduled_at=schedule.next_occurrence
                )
                created.append(meeting)

                schedule.next_occurrence = self._calculate_next_occurrence(
                    schedule.recurrence,
                    from_date=schedule.next_occurrence
                )

        if created:
            self._save()
        return created

    # 조회
    def get_meeting(self, meeting_id: str) -> Optional[Meeting]:
        """미팅 조회"""
        return self._meetings.get(meeting_id)

    def list_meetings(
        self,
        phase: int = None,
        status: MeetingStatus = None,
        meeting_type: MeetingType = None
    ) -> List[Meeting]:
        """미팅 목록 조회"""
        result = list(self._meetings.values())

        if phase is not None:
            result = [m for m in result if m.phase == phase]
        if status is not None:
            result = [m for m in result if m.status == status]
        if meeting_type is not None:
            result = [m for m in result if m.meeting_type == meeting_type]

        return sorted(result, key=lambda m: m.scheduled_at or datetime.min, reverse=True)

    def get_meeting_history(self, team_id: str) -> List[Dict[str, Any]]:
        """팀 미팅 이력"""
        history = []
        for meeting in self._meetings.values():
            if any(p.team_id == team_id for p in meeting.participants):
                history.append({
                    "id": meeting.id,
                    "title": meeting.title,
                    "type": meeting.meeting_type.value,
                    "status": meeting.status.value,
                    "date": meeting.scheduled_at.isoformat() if meeting.scheduled_at else None,
                    "attended": any(p.team_id == team_id and p.attended for p in meeting.participants)
                })
        return sorted(history, key=lambda x: x["date"] or "", reverse=True)

    # 내부 메서드
    def _generate_agenda(
        self,
        meeting_type: MeetingType,
        phase: int,
        teams: List[str]
    ) -> List[AgendaItem]:
        """자동 아젠다 생성"""
        agenda = []

        if meeting_type == MeetingType.STANDUP:
            for i, team in enumerate(teams):
                agenda.append(AgendaItem(
                    id=f"standup-{i}",
                    title=f"{team} 상태 공유",
                    presenter=team,
                    time_minutes=3
                ))

        elif meeting_type == MeetingType.SYNC:
            agenda = [
                AgendaItem(id="1", title="진행 상황 공유", time_minutes=10),
                AgendaItem(id="2", title="블로커 논의", time_minutes=10),
                AgendaItem(id="3", title="다음 주 계획", time_minutes=10),
            ]

        return agenda

    def _generate_summary(self, meeting: Meeting) -> str:
        """자동 요약 생성"""
        discussed = [a.title for a in meeting.agenda if a.status == "discussed"]
        actions = len(meeting.action_items)
        return f"Discussed: {', '.join(discussed)}. Created {actions} action items."

    def _calculate_next_occurrence(
        self,
        recurrence: str,
        from_date: datetime = None
    ) -> Optional[datetime]:
        """다음 발생 시간 계산"""
        base = from_date or datetime.now()

        if recurrence == "daily":
            return base + timedelta(days=1)
        elif recurrence == "weekly":
            return base + timedelta(weeks=1)
        elif recurrence == "none":
            return None
        return None

    def _load(self) -> None:
        """파일에서 로드"""

    def _save(self) -> None:
        """파일에 저장"""

class TeamSyncService:
    """Team Sync 통합 서비스"""

    def __init__(self, project_path: str):
        self.project_path = project_path

        # 컴포넌트 초기화
        self.sync_manager = SyncPointManager(project_path)
        self.board_manager = SharedBoardManager(project_path, self.sync_manager)
        self.meeting_manager = MeetingManager(project_path, self.sync_manager, self.board_manager)

    # Phase 라이프사이클
    def start_phase(self, phase: int, teams: List[str]) -> Dict[str, Any]:
        """Phase 시작"""
        # 1. 보드 생성
        board = self.board_manager.create_board(f"Phase {phase}", phase, teams)

        # 2. Kickoff 미팅 생성
        meeting = self.meeting_manager.create_phase_kickoff(phase, teams)

        # 3. Phase 시작 Sync Point
        sync_point = self.sync_manager.create_barrier(
            f"Phase {phase} Start",
            teams,
            phase=phase
        )

        return {
            "board": board,
            "kickoff_meeting": meeting,
            "sync_point": sync_point
        }

    def end_phase(self, phase: int, teams: List[str]) -> Dict[str, Any]:
        """Phase 종료"""
        # 1. Review 미팅 생성
        meeting = self.meeting_manager.create_phase_review(phase, teams)

        # 2. Phase 종료 Sync Point
        sync_point = self.sync_manager.create_barrier(
            f"Phase {phase} End",
            teams,
            phase=phase
        )

        # 3. 보드 요약
        board = self.board_manager.get_board_by_phase(phase)
        summary = self.board_manager.get_board_summary(board.id) if board else {}

        return {
            "review_meeting": meeting,
            "sync_point": sync_point,
            "board_summary": summary
        }

    # 팀 상태
    def get_team_dashboard(self, team_id: str, phase: int = None) -> Dict[str, Any]:
        """팀 대시보드"""
        result = {
            "team_id": team_id,
            "waiting_sync_points": [],
            "board_status": None,
            "pending_meetings": [],
            "pending_actions": []
        }

        # 대기 중인 Sync Points
        result["waiting_sync_points"] = [
            {"id": sp.id, "name": sp.name, "status": sp.status.value}
            for sp in self.sync_manager.list_waiting_for_team(team_id)
        ]

        # 보드 상태
        if phase:
            board = self.board_manager.get_board_by_phase(phase)
            if board:
                result["board_status"] = self.board_manager.get_team_status(board.id, team_id)

        # 예정된 미팅
        result["pending_meetings"] = [
            {"id": m.id, "title": m.title, "scheduled_at": m.scheduled_at.isoformat() if m.scheduled_at else None}
            for m in self.meeting_manager.list_meetings(phase=phase, status=MeetingStatus.SCHEDULED)
            if any(p.team_id == team_id for p in m.participants)
        ]

        # 미완료 액션
        result["pending_actions"] = self.meeting_manager.get_pending_actions(team_id)

        return result

    # 긴급 동기화
    def create_urgent_sync(
        self,
        title: str,
        teams: List[str],
        blocker_ids: List[str] = None,
        phase: int = 0
    ) -> Dict[str, Any]:
        """긴급 동기화"""
        # 1. Escalation 미팅 생성
        meeting = self.meeting_manager.create_escalation_meeting(
            blocker_ids or [],
            teams,
            phase
        )

        # 2. Sync Point 생성
        sync_point = self.sync_manager.create_barrier(
            f"Urgent: {title}",
            teams,
            phase=phase,
            timeout_minutes=60
        )

        return {
            "meeting": meeting,
            "sync_point": sync_point
        }

    # 전체 상태
    def get_phase_overview(self, phase: int) -> Dict[str, Any]:
        """Phase 전체 현황"""
        board = self.board_manager.get_board_by_phase(phase)

        return {
            "phase": phase,
            "board_summary": self.board_manager.get_board_summary(board.id) if board else None,
            "active_sync_points": len(self.sync_manager.list_active(phase)),
            "blockers": self.board_manager.get_blockers_summary(board.id) if board else [],
            "upcoming_meetings": [
                {"id": m.id, "title": m.title, "type": m.meeting_type.value}
                for m in self.meeting_manager.list_meetings(phase=phase, status=MeetingStatus.SCHEDULED)
            ],
            "dependency_graph": self.board_manager.get_dependency_graph(board.id) if board else None
        }

    def get_all_pending_actions(self) -> List[Dict[str, Any]]:
        """모든 미완료 액션"""
        return self.meeting_manager.get_pending_actions()
