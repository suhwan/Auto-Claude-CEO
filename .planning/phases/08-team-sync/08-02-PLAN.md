from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List, Dict, Any

class TaskStatus(Enum):
    """작업 상태"""
    NOT_STARTED = "not_started"
    IN_PROGRESS = "in_progress"
    BLOCKED = "blocked"
    WAITING = "waiting"
    COMPLETED = "completed"

class BlockerType(Enum):
    """블로커 유형"""
    DEPENDENCY = "dependency"      # 다른 팀 의존성
    TECHNICAL = "technical"        # 기술적 문제
    RESOURCE = "resource"          # 리소스 부족
    DECISION = "decision"          # 결정 대기
    EXTERNAL = "external"          # 외부 요인

@dataclass
class BoardTask:
    """보드 작업 항목"""
    id: str
    team_id: str
    title: str
    description: str = ""
    status: TaskStatus = TaskStatus.NOT_STARTED
    progress: int = 0  # 0-100

    # 의존성
    depends_on: List[str] = field(default_factory=list)  # 다른 task ID
    blocking: List[str] = field(default_factory=list)    # 이 작업이 블로킹하는 task ID

    # 담당자
    assignee: Optional[str] = None
    priority: int = 3  # 1-5

    # 타임스탬프
    created_at: datetime = field(default_factory=datetime.now)
    started_at: Optional[datetime] = None
    completed_at: Optional[datetime] = None

    # 메타데이터
    metadata: Dict[str, Any] = field(default_factory=dict)

@dataclass
class Blocker:
    """블로커"""
    id: str
    team_id: str
    blocker_type: BlockerType
    title: str
    description: str = ""

    # 관련 항목
    related_task_id: Optional[str] = None
    blocking_team_id: Optional[str] = None  # 누가 블로킹하는지

    # 상태
    resolved: bool = False
    resolved_at: Optional[datetime] = None
    resolution: Optional[str] = None

    # 타임스탬프
    created_at: datetime = field(default_factory=datetime.now)
    priority: str = "medium"  # low, medium, high, critical

@dataclass
class TeamLane:
    """팀별 레인"""
    team_id: str
    team_name: str
    display_order: int = 0

    # 작업
    tasks: List[BoardTask] = field(default_factory=list)
    blockers: List[Blocker] = field(default_factory=list)

    # 상태 요약
    total_tasks: int = 0
    completed_tasks: int = 0
    in_progress_tasks: int = 0
    blocked_tasks: int = 0

    # 현재 포커스
    current_focus: Optional[str] = None  # 현재 진행 중인 주요 작업

    # 마지막 업데이트
    last_updated: datetime = field(default_factory=datetime.now)

@dataclass
class DependencyEdge:
    """의존성 엣지"""
    from_team: str
    from_task: str
    to_team: str
    to_task: str
    status: str = "pending"  # pending, satisfied, blocked

@dataclass
class SharedBoard:
    """공유 보드"""
    id: str
    name: str
    phase: int

    # 레인
    lanes: List[TeamLane] = field(default_factory=list)

    # 의존성 그래프
    dependencies: List[DependencyEdge] = field(default_factory=list)

    # Sync Points 연결
    active_sync_points: List[str] = field(default_factory=list)

    # 타임스탬프
    created_at: datetime = field(default_factory=datetime.now)
    last_updated: datetime = field(default_factory=datetime.now)

    # 설정
    auto_sync: bool = True
    update_interval_seconds: int = 60
```

class SharedBoardManager:
    """Shared Board 관리"""

    def __init__(self, project_path: str, sync_manager: SyncPointManager = None):
        self.project_path = project_path
        self.sync_manager = sync_manager or SyncPointManager(project_path)
        self.board_dir = Path(project_path) / ".planning" / "team_sync" / "shared_board"
        self.board_dir.mkdir(parents=True, exist_ok=True)

        self._boards: Dict[str, SharedBoard] = {}
        self._load()

    # 보드 관리
    def create_board(self, name: str, phase: int, teams: List[str]) -> SharedBoard:
        """새 보드 생성"""
        board = SharedBoard(
            id=f"board-{uuid.uuid4().hex[:8]}",
            name=name,
            phase=phase,
            lanes=[
                TeamLane(team_id=t, team_name=t, display_order=i)
                for i, t in enumerate(teams)
            ]
        )
        self._boards[board.id] = board
        self._save()
        return board

    def get_board(self, board_id: str) -> Optional[SharedBoard]:
        """보드 조회"""
        return self._boards.get(board_id)

    def get_board_by_phase(self, phase: int) -> Optional[SharedBoard]:
        """Phase로 보드 조회"""
        for board in self._boards.values():
            if board.phase == phase:
                return board
        return None

    # 작업 관리
    def add_task(
        self,
        board_id: str,
        team_id: str,
        title: str,
        description: str = "",
        depends_on: List[str] = None
    ) -> Optional[BoardTask]:
        """작업 추가"""
        board = self._boards.get(board_id)
        if not board:
            return None

        lane = self._get_lane(board, team_id)
        if not lane:
            return None

        task = BoardTask(
            id=f"task-{uuid.uuid4().hex[:8]}",
            team_id=team_id,
            title=title,
            description=description,
            depends_on=depends_on or []
        )
        lane.tasks.append(task)
        lane.total_tasks += 1

        # 의존성 엣지 생성
        if depends_on:
            for dep_id in depends_on:
                dep_task, dep_lane = self._find_task(board, dep_id)
                if dep_task:
                    board.dependencies.append(DependencyEdge(
                        from_team=dep_lane.team_id,
                        from_task=dep_id,
                        to_team=team_id,
                        to_task=task.id
                    ))

        self._update_lane_stats(lane)
        board.last_updated = datetime.now()
        self._save()
        return task

    def update_task_status(
        self,
        board_id: str,
        task_id: str,
        status: TaskStatus,
        progress: int = None
    ) -> bool:
        """작업 상태 업데이트"""
        board = self._boards.get(board_id)
        if not board:
            return False

        task, lane = self._find_task(board, task_id)
        if not task:
            return False

        old_status = task.status
        task.status = status

        if progress is not None:
            task.progress = progress

        if status == TaskStatus.IN_PROGRESS and old_status == TaskStatus.NOT_STARTED:
            task.started_at = datetime.now()
            lane.current_focus = task.title
        elif status == TaskStatus.COMPLETED:
            task.completed_at = datetime.now()
            task.progress = 100
            self._update_dependencies(board, task_id)

        self._update_lane_stats(lane)
        board.last_updated = datetime.now()
        lane.last_updated = datetime.now()
        self._save()
        return True

    # 블로커 관리
    def add_blocker(
        self,
        board_id: str,
        team_id: str,
        blocker_type: BlockerType,
        title: str,
        description: str = "",
        related_task_id: str = None,
        blocking_team_id: str = None
    ) -> Optional[Blocker]:
        """블로커 추가"""
        board = self._boards.get(board_id)
        if not board:
            return None

        lane = self._get_lane(board, team_id)
        if not lane:
            return None

        blocker = Blocker(
            id=f"blocker-{uuid.uuid4().hex[:8]}",
            team_id=team_id,
            blocker_type=blocker_type,
            title=title,
            description=description,
            related_task_id=related_task_id,
            blocking_team_id=blocking_team_id
        )
        lane.blockers.append(blocker)

        # 관련 작업 상태 업데이트
        if related_task_id:
            self.update_task_status(board_id, related_task_id, TaskStatus.BLOCKED)

        board.last_updated = datetime.now()
        self._save()
        return blocker

    def resolve_blocker(
        self,
        board_id: str,
        blocker_id: str,
        resolution: str = ""
    ) -> bool:
        """블로커 해결"""
        board = self._boards.get(board_id)
        if not board:
            return False

        for lane in board.lanes:
            for blocker in lane.blockers:
                if blocker.id == blocker_id:
                    blocker.resolved = True
                    blocker.resolved_at = datetime.now()
                    blocker.resolution = resolution

                    # 관련 작업 상태 복구
                    if blocker.related_task_id:
                        self.update_task_status(
                            board_id,
                            blocker.related_task_id,
                            TaskStatus.IN_PROGRESS
                        )

                    board.last_updated = datetime.now()
                    self._save()
                    return True
        return False

    # 조회
    def get_team_status(self, board_id: str, team_id: str) -> Dict[str, Any]:
        """팀 상태 조회"""
        board = self._boards.get(board_id)
        if not board:
            return {}

        lane = self._get_lane(board, team_id)
        if not lane:
            return {}

        return {
            "team_id": lane.team_id,
            "total_tasks": lane.total_tasks,
            "completed": lane.completed_tasks,
            "in_progress": lane.in_progress_tasks,
            "blocked": lane.blocked_tasks,
            "progress_percent": (lane.completed_tasks / lane.total_tasks * 100) if lane.total_tasks > 0 else 0,
            "current_focus": lane.current_focus,
            "active_blockers": [b for b in lane.blockers if not b.resolved],
            "last_updated": lane.last_updated.isoformat()
        }

    def get_board_summary(self, board_id: str) -> Dict[str, Any]:
        """보드 요약"""
        board = self._boards.get(board_id)
        if not board:
            return {}

        total_tasks = sum(l.total_tasks for l in board.lanes)
        completed_tasks = sum(l.completed_tasks for l in board.lanes)
        blocked_count = sum(len([b for b in l.blockers if not b.resolved]) for l in board.lanes)

        return {
            "board_id": board.id,
            "name": board.name,
            "phase": board.phase,
            "teams": [l.team_id for l in board.lanes],
            "total_tasks": total_tasks,
            "completed_tasks": completed_tasks,
            "progress_percent": (completed_tasks / total_tasks * 100) if total_tasks > 0 else 0,
            "active_blockers": blocked_count,
            "pending_sync_points": len(board.active_sync_points),
            "last_updated": board.last_updated.isoformat()
        }

    def get_dependency_graph(self, board_id: str) -> Dict[str, Any]:
        """의존성 그래프 조회"""
        board = self._boards.get(board_id)
        if not board:
            return {}

        return {
            "nodes": [
                {"id": l.team_id, "tasks": len(l.tasks)}
                for l in board.lanes
            ],
            "edges": [
                {
                    "from": d.from_team,
                    "to": d.to_team,
                    "status": d.status,
                    "task_from": d.from_task,
                    "task_to": d.to_task
                }
                for d in board.dependencies
            ]
        }

    def get_blockers_summary(self, board_id: str) -> List[Dict[str, Any]]:
        """모든 블로커 요약"""
        board = self._boards.get(board_id)
        if not board:
            return []

        blockers = []
        for lane in board.lanes:
            for b in lane.blockers:
                if not b.resolved:
                    blockers.append({
                        "id": b.id,
                        "team": lane.team_id,
                        "type": b.blocker_type.value,
                        "title": b.title,
                        "blocking_team": b.blocking_team_id,
                        "priority": b.priority,
                        "created_at": b.created_at.isoformat()
                    })
        return sorted(blockers, key=lambda x: x["priority"], reverse=True)

    # 내부 메서드
    def _get_lane(self, board: SharedBoard, team_id: str) -> Optional[TeamLane]:
        """팀 레인 찾기"""
        for lane in board.lanes:
            if lane.team_id == team_id:
                return lane
        return None

    def _find_task(self, board: SharedBoard, task_id: str) -> tuple:
        """작업 찾기"""
        for lane in board.lanes:
            for task in lane.tasks:
                if task.id == task_id:
                    return task, lane
        return None, None

    def _update_lane_stats(self, lane: TeamLane) -> None:
        """레인 통계 업데이트"""
        lane.total_tasks = len(lane.tasks)
        lane.completed_tasks = sum(1 for t in lane.tasks if t.status == TaskStatus.COMPLETED)
        lane.in_progress_tasks = sum(1 for t in lane.tasks if t.status == TaskStatus.IN_PROGRESS)
        lane.blocked_tasks = sum(1 for t in lane.tasks if t.status == TaskStatus.BLOCKED)

    def _update_dependencies(self, board: SharedBoard, completed_task_id: str) -> None:
        """의존성 상태 업데이트"""
        for dep in board.dependencies:
            if dep.from_task == completed_task_id:
                dep.status = "satisfied"

    def _load(self) -> None:
        """파일에서 로드"""

    def _save(self) -> None:
        """파일에 저장"""
```
