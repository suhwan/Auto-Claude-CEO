from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List, Dict, Any

class NotificationType(Enum):
    """알림 유형"""
    CONTRACT_UPDATED = "contract_updated"
    BREAKING_CHANGE = "breaking_change"
    DEPRECATION = "deprecation"
    NEW_CONSUMER = "new_consumer"
    CONSUMER_REMOVED = "consumer_removed"
    DEPENDENCY_CHANGED = "dependency_changed"
    CONTRACT_RETIRED = "contract_retired"

class NotificationPriority(Enum):
    """알림 우선순위"""
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"
    CRITICAL = "critical"

class NotificationStatus(Enum):
    """알림 상태"""
    PENDING = "pending"
    SENT = "sent"
    READ = "read"
    ACKNOWLEDGED = "acknowledged"

@dataclass
class NotificationRecipient:
    """알림 수신자"""
    team_id: str
    team_name: str
    role: str  # owner, consumer, dependent
    status: NotificationStatus = NotificationStatus.PENDING
    sent_at: Optional[datetime] = None
    read_at: Optional[datetime] = None

@dataclass
class Notification:
    """알림"""
    id: str
    notification_type: NotificationType
    priority: NotificationPriority

    # 관련 계약
    contract_id: str
    contract_name: str
    contract_version: str

    # 내용
    title: str
    message: str
    details: Dict[str, Any] = field(default_factory=dict)
    # breaking_changes, affected_endpoints 등

    # 수신자
    recipients: List[NotificationRecipient] = field(default_factory=list)

    # 타임스탬프
    created_at: datetime = field(default_factory=datetime.now)
    expires_at: Optional[datetime] = None

    # 액션
    requires_action: bool = False
    action_deadline: Optional[datetime] = None
    action_items: List[str] = field(default_factory=list)

@dataclass
class NotificationSubscription:
    """알림 구독 설정"""
    team_id: str
    contract_id: Optional[str] = None  # None이면 모든 계약
    notification_types: List[NotificationType] = field(default_factory=list)
    min_priority: NotificationPriority = NotificationPriority.LOW
    enabled: bool = True

class ContractWatcher:
    """Contract 변경 감시"""

    def __init__(self, registry: ContractRegistry, validator: ContractValidator):
        self.registry = registry
        self.validator = validator
        self._watchers: Dict[str, List[Callable]] = {}

    def watch(self, contract_id: str, callback: Callable) -> str:
        """특정 계약 변경 감시"""
        if contract_id not in self._watchers:
            self._watchers[contract_id] = []
        self._watchers[contract_id].append(callback)
        return f"watch-{contract_id}-{len(self._watchers[contract_id])}"

    def unwatch(self, watcher_id: str) -> bool:
        """감시 해제"""

    def detect_changes(
        self,
        old_contract: Contract,
        new_contract: Contract
    ) -> Dict[str, Any]:
        """변경사항 감지"""
        changes = {
            "version_changed": old_contract.version != new_contract.version,
            "is_breaking": False,
            "changes": [],
            "affected_areas": []
        }

        # 버전 변경 분석
        if new_contract.version.major > old_contract.version.major:
            changes["is_breaking"] = True

        # 호환성 검사
        compat = self.validator.check_compatibility(old_contract, new_contract)
        changes["compatibility"] = compat
        if not compat["compatible"]:
            changes["is_breaking"] = True
            changes["breaking_changes"] = compat["breaking_changes"]

        # 상세 변경사항
        changes["additions"] = compat.get("additions", [])
        changes["modifications"] = compat.get("modifications", [])

        return changes

    def analyze_impact(self, contract_id: str, changes: Dict[str, Any]) -> Dict[str, Any]:
        """변경 영향 분석"""
        contract = self.registry.get(contract_id)
        if not contract:
            return {}

        impact = {
            "affected_consumers": [],
            "affected_dependents": [],
            "migration_required": changes.get("is_breaking", False),
            "estimated_impact": "low"
        }

        # 영향받는 Consumer 찾기
        if changes.get("is_breaking"):
            impact["affected_consumers"] = self.validator.find_breaking_consumers(contract_id)
            impact["estimated_impact"] = "high" if impact["affected_consumers"] else "medium"

        # 의존하는 계약 찾기
        for other_id, other_meta in self.registry._index.items():
            other = self.registry.get(other_id)
            if other and contract_id in other.depends_on:
                impact["affected_dependents"].append({
                    "contract_id": other_id,
                    "contract_name": other.name
                })

        return impact

    def notify_change(
        self,
        contract_id: str,
        old_contract: Contract,
        new_contract: Contract
    ) -> List[Notification]:
        """변경 알림 생성 및 발송"""
        changes = self.detect_changes(old_contract, new_contract)
        impact = self.analyze_impact(contract_id, changes)

        notifications = []

        # 변경 유형에 따른 알림 생성
        if changes.get("is_breaking"):
            notif = self._create_breaking_change_notification(
                new_contract, changes, impact
            )
            notifications.append(notif)
        else:
            notif = self._create_update_notification(
                new_contract, changes
            )
            notifications.append(notif)

        # watcher 콜백 호출
        if contract_id in self._watchers:
            for callback in self._watchers[contract_id]:
                callback(new_contract, changes)

        return notifications

    def _create_breaking_change_notification(
        self,
        contract: Contract,
        changes: Dict,
        impact: Dict
    ) -> Notification:
        """Breaking Change 알림 생성"""

    def _create_update_notification(
        self,
        contract: Contract,
        changes: Dict
    ) -> Notification:
        """일반 업데이트 알림 생성"""
```

class NotificationService:
    """알림 서비스"""

    def __init__(self, project_path: str, registry: ContractRegistry):
        self.project_path = project_path
        self.registry = registry
        self.notifications_dir = Path(project_path) / ".planning" / "contracts" / "notifications"
        self.notifications_dir.mkdir(parents=True, exist_ok=True)

        self._notifications: List[Notification] = []
        self._subscriptions: Dict[str, List[NotificationSubscription]] = {}
        self._load()

    # 알림 생성 및 발송
    def send(self, notification: Notification) -> bool:
        """알림 발송"""
        # 1. 알림 저장
        self._notifications.append(notification)

        # 2. 수신자별 상태 업데이트
        for recipient in notification.recipients:
            recipient.status = NotificationStatus.SENT
            recipient.sent_at = datetime.now()

        # 3. 저장
        self._save()

        return True

    def send_to_team(
        self,
        team_id: str,
        notification_type: NotificationType,
        contract: Contract,
        message: str,
        priority: NotificationPriority = NotificationPriority.MEDIUM
    ) -> Notification:
        """특정 팀에 알림 발송"""
        notif = Notification(
            id=f"notif-{uuid.uuid4().hex[:8]}",
            notification_type=notification_type,
            priority=priority,
            contract_id=contract.id,
            contract_name=contract.name,
            contract_version=str(contract.version),
            title=self._generate_title(notification_type, contract),
            message=message,
            recipients=[
                NotificationRecipient(
                    team_id=team_id,
                    team_name=team_id,
                    role="recipient"
                )
            ]
        )
        self.send(notif)
        return notif

    def broadcast_to_consumers(
        self,
        contract: Contract,
        notification_type: NotificationType,
        message: str,
        priority: NotificationPriority = NotificationPriority.MEDIUM
    ) -> Notification:
        """모든 Consumer에게 알림 발송"""
        recipients = [
            NotificationRecipient(
                team_id=c.team_id,
                team_name=c.team_name,
                role="consumer"
            )
            for c in contract.consumers
        ]

        # 소유자에게도 발송
        recipients.append(
            NotificationRecipient(
                team_id=contract.owner.team_id,
                team_name=contract.owner.team_name,
                role="owner"
            )
        )

        notif = Notification(
            id=f"notif-{uuid.uuid4().hex[:8]}",
            notification_type=notification_type,
            priority=priority,
            contract_id=contract.id,
            contract_name=contract.name,
            contract_version=str(contract.version),
            title=self._generate_title(notification_type, contract),
            message=message,
            recipients=recipients
        )
        self.send(notif)
        return notif

    # 조회
    def get_notifications(
        self,
        team_id: str = None,
        status: NotificationStatus = None,
        notification_type: NotificationType = None
    ) -> List[Notification]:
        """알림 조회"""
        result = self._notifications

        if team_id:
            result = [
                n for n in result
                if any(r.team_id == team_id for r in n.recipients)
            ]

        if status:
            result = [
                n for n in result
                if any(r.status == status for r in n.recipients)
            ]

        if notification_type:
            result = [n for n in result if n.notification_type == notification_type]

        return sorted(result, key=lambda n: n.created_at, reverse=True)

    def get_unread(self, team_id: str) -> List[Notification]:
        """읽지 않은 알림"""
        return self.get_notifications(team_id, NotificationStatus.SENT)

    def get_pending_actions(self, team_id: str) -> List[Notification]:
        """조치 필요한 알림"""
        return [
            n for n in self.get_notifications(team_id)
            if n.requires_action and not self._is_acknowledged(n, team_id)
        ]

    # 상태 업데이트
    def mark_read(self, notification_id: str, team_id: str) -> bool:
        """읽음 표시"""
        for notif in self._notifications:
            if notif.id == notification_id:
                for recipient in notif.recipients:
                    if recipient.team_id == team_id:
                        recipient.status = NotificationStatus.READ
                        recipient.read_at = datetime.now()
                        self._save()
                        return True
        return False

    def acknowledge(self, notification_id: str, team_id: str) -> bool:
        """조치 완료 표시"""
        for notif in self._notifications:
            if notif.id == notification_id:
                for recipient in notif.recipients:
                    if recipient.team_id == team_id:
                        recipient.status = NotificationStatus.ACKNOWLEDGED
                        self._save()
                        return True
        return False

    # 구독 관리
    def subscribe(self, subscription: NotificationSubscription) -> bool:
        """알림 구독"""

    def unsubscribe(self, team_id: str, contract_id: str = None) -> bool:
        """구독 해제"""

    def get_subscriptions(self, team_id: str) -> List[NotificationSubscription]:
        """구독 목록"""

    # 통계
    def get_stats(self, team_id: str = None) -> Dict[str, Any]:
        """알림 통계"""
        notifications = self.get_notifications(team_id)
        return {
            "total": len(notifications),
            "unread": len([n for n in notifications if self._has_unread(n, team_id)]),
            "pending_actions": len(self.get_pending_actions(team_id) if team_id else []),
            "by_type": self._count_by_type(notifications),
            "by_priority": self._count_by_priority(notifications)
        }

    def _load(self) -> None:
        """파일에서 로드"""

    def _save(self) -> None:
        """파일에 저장"""

    def _generate_title(self, ntype: NotificationType, contract: Contract) -> str:
        """알림 제목 생성"""
```

class TeamContractService:
    """Team Contract 통합 서비스"""

    def __init__(self, project_path: str):
        self.project_path = project_path

        # 컴포넌트 초기화
        self.registry = ContractRegistry(project_path)
        self.validator = ContractValidator(self.registry)
        self.version_manager = ContractVersionManager(self.registry)
        self.watcher = ContractWatcher(self.registry, self.validator)
        self.notifications = NotificationService(project_path, self.registry)

    # 통합 API
    def create_contract(self, contract: Contract) -> Dict[str, Any]:
        """계약 생성"""
        # 1. 유효성 검사
        errors = self.validator.validate(contract)
        if errors:
            return {"success": False, "errors": errors}

        # 2. 의존성 검사
        deps = self.validator.check_dependencies(contract)
        if not deps["satisfied"]:
            return {"success": False, "errors": deps["missing"]}

        # 3. 등록
        self.registry.register(contract)

        return {"success": True, "contract": contract}

    def update_contract(
        self,
        contract_id: str,
        changes: Dict[str, Any],
        bump_type: str = "patch",
        changelog: str = ""
    ) -> Dict[str, Any]:
        """계약 업데이트"""
        old_contract = self.registry.get(contract_id)
        if not old_contract:
            return {"success": False, "error": "Contract not found"}

        # 1. 업데이트
        new_contract = self.version_manager.update(
            contract_id, changes, bump_type, changelog
        )

        # 2. 변경 알림
        notifications = self.watcher.notify_change(
            contract_id, old_contract, new_contract
        )

        # 3. 알림 발송
        for notif in notifications:
            self.notifications.send(notif)

        return {
            "success": True,
            "contract": new_contract,
            "notifications_sent": len(notifications)
        }

    def deprecate_contract(self, contract_id: str, reason: str) -> Dict[str, Any]:
        """계약 폐기 예정"""
        contract = self.registry.get(contract_id)
        if not contract:
            return {"success": False, "error": "Contract not found"}

        # 상태 변경
        self.registry.update_status(contract_id, ContractStatus.DEPRECATED)

        # 알림 발송
        self.notifications.broadcast_to_consumers(
            contract,
            NotificationType.DEPRECATION,
            f"Contract '{contract.name}' will be deprecated. Reason: {reason}",
            NotificationPriority.HIGH
        )

        return {"success": True}

    def add_consumer(self, contract_id: str, team_id: str, team_name: str) -> Dict[str, Any]:
        """Consumer 추가"""
        contract = self.registry.get(contract_id)
        if not contract:
            return {"success": False, "error": "Contract not found"}

        consumer = ContractConsumer(
            team_id=team_id,
            team_name=team_name,
            version_used=contract.version
        )
        self.registry.add_consumer(contract_id, consumer)

        # 소유자에게 알림
        self.notifications.send_to_team(
            contract.owner.team_id,
            NotificationType.NEW_CONSUMER,
            contract,
            f"Team '{team_name}' started using your contract.",
            NotificationPriority.LOW
        )

        return {"success": True, "consumer": consumer}

    def get_team_dashboard(self, team_id: str) -> Dict[str, Any]:
        """팀 대시보드 정보"""
        return {
            "owned_contracts": self.registry.get_contracts_by_owner(team_id),
            "consumed_contracts": self.registry.get_contracts_by_consumer(team_id),
            "unread_notifications": len(self.notifications.get_unread(team_id)),
            "pending_actions": len(self.notifications.get_pending_actions(team_id)),
            "notification_stats": self.notifications.get_stats(team_id)
        }

    def get_contract_health(self, contract_id: str) -> Dict[str, Any]:
        """계약 상태 정보"""
        contract = self.registry.get(contract_id)
        if not contract:
            return {"error": "Contract not found"}

        return {
            "contract": contract,
            "consumers": contract.consumers,
            "version_history": self.version_manager.list_versions(contract_id),
            "dependencies": self.validator.check_dependencies(contract),
            "validation": self.validator.validate(contract)
        }
