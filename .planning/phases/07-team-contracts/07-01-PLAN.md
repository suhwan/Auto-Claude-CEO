```python
from enum import Enum
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional, List, Dict, Any

class ContractType(Enum):
    """계약 유형"""
    API = "api"                    # REST/GraphQL API
    UI = "ui"                      # UI 컴포넌트
    DATA = "data"                  # 데이터 모델
    EVENT = "event"                # 이벤트/메시지
    CONFIG = "config"              # 설정 스키마

class ContractStatus(Enum):
    """계약 상태"""
    DRAFT = "draft"                # 초안
    PROPOSED = "proposed"          # 제안됨
    APPROVED = "approved"          # 승인됨
    DEPRECATED = "deprecated"      # 폐기 예정
    RETIRED = "retired"            # 폐기됨

@dataclass
class ContractVersion:
    """버전 정보"""
    major: int = 1
    minor: int = 0
    patch: int = 0

    def __str__(self) -> str:
        return f"{self.major}.{self.minor}.{self.patch}"

    def is_compatible_with(self, other: 'ContractVersion') -> bool:
        """호환성 검사 (major 버전 동일 시 호환)"""
        return self.major == other.major

    def bump_major(self) -> 'ContractVersion':
        return ContractVersion(self.major + 1, 0, 0)

    def bump_minor(self) -> 'ContractVersion':
        return ContractVersion(self.major, self.minor + 1, 0)

    def bump_patch(self) -> 'ContractVersion':
        return ContractVersion(self.major, self.minor, self.patch + 1)

@dataclass
class ContractOwner:
    """계약 소유자"""
    team_id: str
    team_name: str
    contact: Optional[str] = None

@dataclass
class ContractConsumer:
    """계약 사용자"""
    team_id: str
    team_name: str
    version_used: ContractVersion = field(default_factory=ContractVersion)
    registered_at: datetime = field(default_factory=datetime.now)

@dataclass
class Contract:
    """팀 간 공유 계약"""
    id: str
    name: str
    contract_type: ContractType
    version: ContractVersion
    status: ContractStatus

    # 소유 및 사용
    owner: ContractOwner
    consumers: List[ContractConsumer] = field(default_factory=list)

    # 스키마 정의
    schema: Dict[str, Any] = field(default_factory=dict)
    # JSON Schema 형식
    # API: { endpoints: [...], request_schema: {...}, response_schema: {...} }
    # UI: { props: {...}, events: [...], slots: [...] }
    # DATA: { properties: {...}, required: [...] }

    # 메타데이터
    description: str = ""
    tags: List[str] = field(default_factory=list)
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)

    # 버전 히스토리
    changelog: List[Dict[str, Any]] = field(default_factory=list)
    # [{ version: "1.0.0", changes: [...], date: "...", breaking: false }]

    # 의존성
    depends_on: List[str] = field(default_factory=list)  # 다른 contract ID 목록
```

```python
@dataclass
class APIEndpoint:
    """API 엔드포인트 정의"""
    path: str                          # /api/v1/users/{id}
    method: str                        # GET, POST, PUT, DELETE
    description: str = ""

    # 파라미터
    path_params: Dict[str, Any] = field(default_factory=dict)
    query_params: Dict[str, Any] = field(default_factory=dict)
    headers: Dict[str, Any] = field(default_factory=dict)

    # 요청/응답
    request_body: Optional[Dict[str, Any]] = None
    response: Dict[str, Dict[str, Any]] = field(default_factory=dict)
    # { "200": { schema: {...} }, "400": { schema: {...} } }

    # 인증/권한
    auth_required: bool = True
    permissions: List[str] = field(default_factory=list)

@dataclass
class APIContract(Contract):
    """API 전용 계약"""
    base_url: str = ""
    endpoints: List[APIEndpoint] = field(default_factory=list)

    # 공통 스키마 (재사용)
    shared_schemas: Dict[str, Dict[str, Any]] = field(default_factory=dict)
    # { "User": { type: "object", properties: {...} } }

    # 에러 코드 정의
    error_codes: Dict[str, str] = field(default_factory=dict)
    # { "USER_NOT_FOUND": "User with given ID not found" }
```

```python
@dataclass
class PropDefinition:
    """UI 컴포넌트 prop 정의"""
    name: str
    prop_type: str  # string, number, boolean, object, array
    required: bool = False
    default: Any = None
    description: str = ""
    validation: Optional[Dict[str, Any]] = None

@dataclass
class EventDefinition:
    """UI 이벤트 정의"""
    name: str
    payload: Dict[str, Any] = field(default_factory=dict)
    description: str = ""

@dataclass
class SlotDefinition:
    """UI 슬롯 정의"""
    name: str
    description: str = ""
    scope_props: Dict[str, str] = field(default_factory=dict)

@dataclass
class UIContract(Contract):
    """UI 컴포넌트 계약"""
    component_name: str = ""
    props: List[PropDefinition] = field(default_factory=list)
    events: List[EventDefinition] = field(default_factory=list)
    slots: List[SlotDefinition] = field(default_factory=list)

    # 스타일 가이드
    css_classes: List[str] = field(default_factory=list)
    theme_variables: Dict[str, str] = field(default_factory=dict)

@dataclass
class FieldDefinition:
    """데이터 필드 정의"""
    name: str
    field_type: str  # string, number, boolean, date, object, array
    required: bool = False
    nullable: bool = True
    description: str = ""
    constraints: Dict[str, Any] = field(default_factory=dict)
    # { minLength: 1, maxLength: 100, pattern: "^[a-z]+$" }

@dataclass
class DataContract(Contract):
    """데이터 모델 계약"""
    model_name: str = ""
    fields: List[FieldDefinition] = field(default_factory=list)

    # 관계
    relations: Dict[str, str] = field(default_factory=dict)
    # { "author": "User", "comments": "Comment[]" }

    # 인덱스
    indexes: List[List[str]] = field(default_factory=list)
    # [ ["email"], ["created_at", "status"] ]

    # 검증 규칙
    validation_rules: List[str] = field(default_factory=list)
    # [ "end_date > start_date", "quantity >= 0" ]
```

```python
class ContractSerializer:
    """Contract JSON 직렬화"""

    @staticmethod
    def version_to_dict(version: ContractVersion) -> dict:
        return {"major": version.major, "minor": version.minor, "patch": version.patch}

    @staticmethod
    def version_from_dict(data: dict) -> ContractVersion:
        return ContractVersion(data["major"], data["minor"], data["patch"])

    def contract_to_dict(contract: Contract) -> dict:
```
