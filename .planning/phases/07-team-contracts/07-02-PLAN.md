class ContractVersionManager:
    """Contract 버전 관리"""

    def __init__(self, registry: ContractRegistry):
        self.registry = registry

    def update(
        self,
        contract_id: str,
        changes: Dict[str, Any],
        bump_type: str = "patch",  # major, minor, patch
        changelog_entry: str = ""
    ) -> Optional[Contract]:
        """계약 업데이트 (새 버전 생성)"""
        contract = self.registry.get(contract_id)
        if not contract:
            return None

        # 1. 현재 버전 히스토리에 저장
        self._archive_version(contract)

        # 2. 버전 범프
        old_version = contract.version
        if bump_type == "major":
            contract.version = old_version.bump_major()
        elif bump_type == "minor":
            contract.version = old_version.bump_minor()
        else:
            contract.version = old_version.bump_patch()

        # 3. 변경사항 적용
        for key, value in changes.items():
            if hasattr(contract, key):
                setattr(contract, key, value)

        # 4. changelog 추가
        contract.changelog.append({
            "version": str(contract.version),
            "previous_version": str(old_version),
            "changes": changelog_entry,
            "date": datetime.now().isoformat(),
            "breaking": bump_type == "major"
        })

        contract.updated_at = datetime.now()

        # 5. 저장
        self.registry._save_contract(contract)

        return contract

    def get_version(self, contract_id: str, version: str) -> Optional[Contract]:
        """특정 버전 조회"""
        # 히스토리에서 로드

        """변경 이력 조회"""

    def rollback(self, contract_id: str, target_version: str) -> Optional[Contract]:

class ContractValidator:
    """Contract 유효성 검사"""

    def __init__(self, registry: ContractRegistry):
        self.registry = registry

    def validate(self, contract: Contract) -> List[str]:
        """계약 유효성 검사, 에러 목록 반환"""
        errors = []

        # 필수 필드 검사
        if not contract.id:
            errors.append("Contract ID is required")
        if not contract.name:
            errors.append("Contract name is required")
        if not contract.owner:
            errors.append("Contract owner is required")

        # 타입별 추가 검사
        if isinstance(contract, APIContract):
            errors.extend(self._validate_api_contract(contract))
        elif isinstance(contract, UIContract):
            errors.extend(self._validate_ui_contract(contract))
        elif isinstance(contract, DataContract):
            errors.extend(self._validate_data_contract(contract))

        return errors

    def is_valid(self, contract: Contract) -> bool:
        """유효성 통과 여부"""
        return len(self.validate(contract)) == 0

    def check_compatibility(
        self,
        old_contract: Contract,
        new_contract: Contract
    ) -> Dict[str, Any]:
        """호환성 검사"""
        result = {
            "compatible": True,
            "breaking_changes": [],
            "additions": [],
            "modifications": []
        }

        # 버전 호환성
        if not new_contract.version.is_compatible_with(old_contract.version):
            result["compatible"] = False
            result["breaking_changes"].append(
                f"Major version changed: {old_contract.version} → {new_contract.version}"
            )

        # 스키마 호환성 (타입별)
        if isinstance(old_contract, APIContract) and isinstance(new_contract, APIContract):
            self._check_api_compatibility(old_contract, new_contract, result)
        elif isinstance(old_contract, DataContract) and isinstance(new_contract, DataContract):
            self._check_data_compatibility(old_contract, new_contract, result)

        return result

    def check_dependencies(self, contract: Contract) -> Dict[str, Any]:
        """의존성 검사"""
        result = {
            "satisfied": True,
            "missing": [],
            "version_mismatch": []
        }

        for dep_id in contract.depends_on:
            dep_contract = self.registry.get(dep_id)
            if not dep_contract:
                result["satisfied"] = False
                result["missing"].append(dep_id)
            elif dep_contract.status == ContractStatus.RETIRED:
                result["satisfied"] = False
                result["version_mismatch"].append(
                    f"{dep_id}: Contract is retired"
                )

        return result

    def find_breaking_consumers(self, contract_id: str) -> List[ContractConsumer]:
        """breaking change 시 영향받는 사용자 찾기"""
        contract = self.registry.get(contract_id)
        if not contract:
            return []

        affected = []
        for consumer in contract.consumers:
            if not contract.version.is_compatible_with(consumer.version_used):
                affected.append(consumer)

        return affected

    def _validate_api_contract(self, contract: APIContract) -> List[str]:
        """API 계약 검증"""
        errors = []
        for endpoint in contract.endpoints:
            if not endpoint.path:
                errors.append(f"Endpoint path is required")
            if endpoint.method not in ["GET", "POST", "PUT", "PATCH", "DELETE"]:
                errors.append(f"Invalid method: {endpoint.method}")
        return errors

    def _validate_ui_contract(self, contract: UIContract) -> List[str]:
        """UI 계약 검증"""

    def _validate_data_contract(self, contract: DataContract) -> List[str]:
        """Data 계약 검증"""

    def _check_api_compatibility(
        self,
        old: APIContract,
        new: APIContract,
        result: Dict
    ) -> None:
        """API 호환성 상세 검사"""

    def _check_data_compatibility(

    ) -> None:
