export interface GsdGoal {

  title: string;
  description: string;

  status: 'active' | 'achieved' | 'abandoned';
  phase?: number;
}

export interface GsdDecision {
  id: string;
  title: string;
  description: string;
  rationale: string;
  made_at: string;
  phase?: number;
}

export interface GsdPattern {
  id: string;
  name: string;
  description: string;
  category: string;
  occurrences: number;
}

export interface GsdMistake {
  id: string;
  description: string;
  impact: string;
  lesson_learned: string;
  occurred_at: string;
  phase?: number;
}

export interface GsdRisk {
  id: string;
  title: string;
  description: string;
  severity: 'low' | 'medium' | 'high' | 'critical';
  likelihood: 'unlikely' | 'possible' | 'likely' | 'certain';
  mitigation?: string;
  status: 'identified' | 'mitigated' | 'occurred' | 'closed';
}

export interface GsdLeaderContext {
  project_id: string;
  phase: number;
  goals: GsdGoal[];
  decisions: GsdDecision[];
  patterns: GsdPattern[];
  mistakes: GsdMistake[];
  risks: GsdRisk[];
  updated_at: string;
}

async getLeaderContext(phase?: number): Promise<GsdLeaderContext | null> {
  const contextDir = path.join(this.projectPath, '.planning', 'leader_context');

  if (!fs.existsSync(contextDir)) {
    return null;
  }

  // Find context file
  const files = fs.readdirSync(contextDir).filter(f => f.endsWith('.json'));

  if (files.length === 0) {
    return null;
  }

  // Get latest context file
  const latestFile = files
    .map(f => ({ name: f, mtime: fs.statSync(path.join(contextDir, f)).mtime }))
    .sort((a, b) => b.mtime.getTime() - a.mtime.getTime())[0];

  const content = fs.readFileSync(path.join(contextDir, latestFile.name), 'utf-8');
  const data = JSON.parse(content);

  return this.transformContextData(data);
}

private transformContextData(data: any): GsdLeaderContext {
  return {
    project_id: data.project_id || 'unknown',
    phase: data.phase || 0,
    goals: (data.goals || []).map((g: any) => ({
      id: g.id,
      title: g.title,
      description: g.description || '',
      priority: g.priority || 'secondary',
      status: g.status || 'active',
      phase: g.phase
    })),
    decisions: (data.decisions || []).map((d: any) => ({
      id: d.id,
      title: d.title,
      description: d.description || '',
      rationale: d.rationale || '',
      made_at: d.made_at || d.created_at || '',
      phase: d.phase
    })),
    patterns: (data.patterns || []).map((p: any) => ({
      id: p.id,
      name: p.name,
      description: p.description || '',
      category: p.category || 'general',
      occurrences: p.occurrences || 1
    })),
    mistakes: (data.mistakes || []).map((m: any) => ({
      id: m.id,
      description: m.description,
      impact: m.impact || '',
      lesson_learned: m.lesson_learned || '',
      occurred_at: m.occurred_at || m.created_at || '',
      phase: m.phase
    })),
    risks: (data.risks || []).map((r: any) => ({
      id: r.id,
      title: r.title,
      description: r.description || '',
      severity: r.severity || 'medium',
      likelihood: r.likelihood || 'possible',
      mitigation: r.mitigation,
      status: r.status || 'identified'
    })),
    updated_at: data.updated_at || new Date().toISOString()
  };
}

ipcMain.handle(
  IPC_CHANNELS.GSD_GET_LEADER_CONTEXT,
  async (_event: IpcMainInvokeEvent, projectPath: string, phase?: number): Promise<IPCResult> => {
    try {
      logger.info('gsd:getLeaderContext', { projectPath, phase });
      const gsdService = getGsdService(projectPath);
      const context = await gsdService.getLeaderContext(phase);
      return { success: true, data: context };
    } catch (error) {
      logger.error('gsd:getLeaderContext failed:', error);
      return { success: false, error: error instanceof Error ? error.message : 'Unknown error' };
    }
  }
);

function GoalsCard({ goals }: { goals: GsdGoal[] }) {
  const activeGoals = goals.filter(g => g.status === 'active');

  return (
    <Card>
      <CardHeader className="pb-2">
        <CardTitle className="text-sm flex items-center gap-2">
          <Target className="h-4 w-4 text-blue-500" />
          Goals
          <Badge variant="outline" className="ml-auto text-xs">
            {activeGoals.length} active
          </Badge>
        </CardTitle>
      </CardHeader>
      <CardContent className="space-y-2">
        {goals.length === 0 ? (
          <p className="text-xs text-muted-foreground">No goals defined</p>
        ) : (
          goals.slice(0, 5).map((goal) => (
            <div key={goal.id} className="flex items-start gap-2">
              <Badge
                variant={goal.priority === 'primary' ? 'default' : 'secondary'}
                className="text-[10px] shrink-0"
              >
                {goal.priority}
              </Badge>
              <div className="flex-1 min-w-0">
                <p className="text-xs font-medium truncate">{goal.title}</p>
                {goal.status === 'achieved' && (
                  <CheckCircle2 className="h-3 w-3 text-green-500 inline ml-1" />
                )}
              </div>
            </div>
          ))
        )}
      </CardContent>
    </Card>
  );
}

function DecisionsCard({ decisions }: { decisions: GsdDecision[] }) {
  const recentDecisions = decisions.slice(-5).reverse();

  return (
    <Card>
      <CardHeader className="pb-2">
        <CardTitle className="text-sm flex items-center gap-2">

          Decisions
          <Badge variant="outline" className="ml-auto text-xs">

          recentDecisions.map((decision) => (
            <div key={decision.id} className="text-xs">
              <p className="font-medium truncate">{decision.title}</p>

                <p className="text-muted-foreground truncate">{decision.rationale}</p>
              )}
            </div>
          ))
        )}
      </CardContent>
    </Card>
  );
}

function RisksCard({ risks }: { risks: GsdRisk[] }) {
  const activeRisks = risks.filter(r => r.status === 'identified');

  const getSeverityColor = (severity: string) => {
    switch (severity) {
      case 'critical': return 'text-red-600 bg-red-50';
      case 'high': return 'text-orange-600 bg-orange-50';
      case 'medium': return 'text-yellow-600 bg-yellow-50';
      default: return 'text-gray-600 bg-gray-50';
    }
  };

  return (
    <Card>
      <CardHeader className="pb-2">
        <CardTitle className="text-sm flex items-center gap-2">
          <AlertTriangle className="h-4 w-4 text-orange-500" />
          Risks
          <Badge variant="outline" className="ml-auto text-xs">
            {activeRisks.length} active
          </Badge>
        </CardTitle>
      </CardHeader>
      <CardContent className="space-y-2">
        {risks.length === 0 ? (
          <p className="text-xs text-muted-foreground">No risks identified</p>
        ) : (
          risks.slice(0, 5).map((risk) => (
            <div key={risk.id} className="flex items-start gap-2">
              <Badge
                variant="outline"
                className={cn('text-[10px] shrink-0', getSeverityColor(risk.severity))}
              >
                {risk.severity}
              </Badge>
              <p className="text-xs truncate flex-1">{risk.title}</p>
            </div>
          ))
        )}
      </CardContent>
    </Card>
  );
}

function PatternsCard({ patterns }: { patterns: GsdPattern[] }) {
  return (
    <Card>
      <CardHeader className="pb-2">
        <CardTitle className="text-sm flex items-center gap-2">
          <RefreshCw className="h-4 w-4 text-green-500" />

          <Badge variant="outline" className="ml-auto text-xs">

          </Badge>
        </CardTitle>
      </CardHeader>
      <CardContent className="space-y-2">

}
