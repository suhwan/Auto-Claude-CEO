export type VerificationStatus = 'pending' | 'approved' | 'rejected' | 'not_required';

export interface GsdVerificationItem {

  description: string;
  checked: boolean;

export interface GsdPlanVerification {
  plan_id: string;
  plan_name: string;
  phase: number;
  auto_qa_passed: boolean;
  verification_status: VerificationStatus;
  checklist: GsdVerificationItem[];
  feedback?: string;
  verified_at?: string;
  verified_by?: string;
}

export interface GsdVerificationHistory {
  id: string;

  feedback?: string;
  timestamp: string;

async getPendingVerifications(): Promise<GsdPlanVerification[]> {
  const stateFile = path.join(this.projectPath, '.planning', 'STATE.md');

  if (!fs.existsSync(stateFile)) {
    return [];
  }

  const content = fs.readFileSync(stateFile, 'utf-8');

  // Parse recent completed plans that need verification
  const recentPlansMatch = /Recent Trend:[\s\S]*?Last \d+ plans: (.+)/i.exec(content);
  if (!recentPlansMatch) {
    return [];
  }

  const recentPlans = recentPlansMatch[1].split(',').map(s => s.trim());
  const completedPlans = recentPlans.filter(p => p.includes('✓'));

  // Check verification status file
  const verificationFile = path.join(this.projectPath, '.planning', 'verifications.json');
  let verifications: Record<string, GsdPlanVerification> = {};

  if (fs.existsSync(verificationFile)) {
    verifications = JSON.parse(fs.readFileSync(verificationFile, 'utf-8'));
  }

  // Find plans needing verification
  const pending: GsdPlanVerification[] = [];

  for (const planRef of completedPlans.slice(-3)) {
    const planId = planRef.replace('✓', '').trim();

    if (!verifications[planId] || verifications[planId].verification_status === 'pending') {
      pending.push({
        plan_id: planId,
        plan_name: await this.getPlanName(planId),
        phase: parseInt(planId.split('-')[0]) || 0,
        auto_qa_passed: true,
        verification_status: 'pending',
        checklist: await this.getDefaultChecklist(planId)
      });
    }
  }

  return pending;
}

private async getPlanName(planId: string): Promise<string> {
  // Try to find plan file
  const [phaseNum, planNum] = planId.split('-');
  const phaseDir = this.findPhaseDir(parseInt(phaseNum));

  if (!phaseDir) return planId;

  const planFile = path.join(phaseDir, `${planId}-PLAN.md`);
  if (!fs.existsSync(planFile)) return planId;

  const content = fs.readFileSync(planFile, 'utf-8');
  const nameMatch = /name:\s*(.+)/i.exec(content);

  return nameMatch?.[1] || planId;
}

private async getDefaultChecklist(planId: string): Promise<GsdVerificationItem[]> {
  // Try to get success_criteria from plan
  const [phaseNum, planNum] = planId.split('-');
  const phaseDir = this.findPhaseDir(parseInt(phaseNum));

  if (!phaseDir) {
    return [
      { id: '1', description: 'Feature works as expected', checked: false },
      { id: '2', description: 'UI displays correctly', checked: false },
      { id: '3', description: 'No visual bugs', checked: false }
    ];
  }

  const planFile = path.join(phaseDir, `${planId}-PLAN.md`);
  if (!fs.existsSync(planFile)) {
    return [
      { id: '1', description: 'Feature works as expected', checked: false },
      { id: '2', description: 'UI displays correctly', checked: false },
      { id: '3', description: 'No visual bugs', checked: false }
    ];
  }

  const content = fs.readFileSync(planFile, 'utf-8');
  const criteriaMatch = /<success_criteria>([\s\S]*?)<\/success_criteria>/i.exec(content);

  if (!criteriaMatch) {
    return [
      { id: '1', description: 'Feature works as expected', checked: false },
      { id: '2', description: 'UI displays correctly', checked: false },
      { id: '3', description: 'No visual bugs', checked: false }
    ];
  }

  const criteria = criteriaMatch[1]
    .split('\n')
    .filter(line => line.includes('[ ]') || line.includes('[x]'))
    .map((line, index) => ({
      id: String(index + 1),
      description: line.replace(/^-\s*\[.\]\s*/, '').trim(),
      checked: false
    }));

  return criteria.length > 0 ? criteria : [
    { id: '1', description: 'Feature works as expected', checked: false },
    { id: '2', description: 'UI displays correctly', checked: false },
    { id: '3', description: 'No visual bugs', checked: false }
  ];
}

async submitVerification(
  planId: string,
  approved: boolean,
  feedback?: string,
  checklist?: GsdVerificationItem[]
): Promise<{ success: boolean; error?: string }> {
  try {
    const verificationFile = path.join(this.projectPath, '.planning', 'verifications.json');
    let verifications: Record<string, any> = {};

    if (fs.existsSync(verificationFile)) {
      verifications = JSON.parse(fs.readFileSync(verificationFile, 'utf-8'));
    }

    verifications[planId] = {
      plan_id: planId,
      verification_status: approved ? 'approved' : 'rejected',
      feedback,
      checklist,
      verified_at: new Date().toISOString()
    };

    // Ensure directory exists
    const dir = path.dirname(verificationFile);
    if (!fs.existsSync(dir)) {
      fs.mkdirSync(dir, { recursive: true });
    }

    fs.writeFileSync(verificationFile, JSON.stringify(verifications, null, 2));

    // If rejected, create a fix request file
    if (!approved && feedback) {
      await this.createFixRequest(planId, feedback);
    }

    return { success: true };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error'
    };
  }
}

private async createFixRequest(planId: string, feedback: string): Promise<void> {
  const fixRequestFile = path.join(
    this.projectPath,
    '.planning',
    'fix_requests',
    `${planId}-FIX-REQUEST.md`
  );

  const dir = path.dirname(fixRequestFile);
  if (!fs.existsSync(dir)) {
    fs.mkdirSync(dir, { recursive: true });
  }

  const content = `# Fix Request: ${planId}

## Status
- Created: ${new Date().toISOString()}
- Status: pending

## User Feedback
${feedback}

## Action Required
Run \`/gsd:plan-fix ${planId}\` to create a fix plan.
`;

  fs.writeFileSync(fixRequestFile, content);
}

interface VerificationPanelProps {
  verification: GsdPlanVerification;
  onApprove: (planId: string, feedback?: string, checklist?: GsdVerificationItem[]) => Promise<void>;
  onReject: (planId: string, feedback: string, checklist?: GsdVerificationItem[]) => Promise<void>;
}

export function VerificationPanel({ verification, onApprove, onReject }: VerificationPanelProps) {
  const [checklist, setChecklist] = useState<GsdVerificationItem[]>(verification.checklist);
  const [feedback, setFeedback] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const allChecked = checklist.every(item => item.checked);
  const someChecked = checklist.some(item => item.checked);

  const toggleItem = (id: string) => {
    setChecklist(prev =>
      prev.map(item =>
        item.id === id ? { ...item, checked: !item.checked } : item
      )
    );
  };

  const handleApprove = async () => {
    setSubmitting(true);
    setError(null);
    try {
      await onApprove(verification.plan_id, feedback || undefined, checklist);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to approve');
    } finally {
      setSubmitting(false);
    }
  };

  const handleReject = async () => {
    if (!feedback.trim()) {
      setError('Please provide feedback for rejection');
      return;
    }
    setSubmitting(true);
    setError(null);
    try {
      await onReject(verification.plan_id, feedback, checklist);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to reject');
    } finally {
      setSubmitting(false);
    }
  };

  return (
    <Card className="border-blue-200 bg-blue-50/30">
      <CardHeader className="pb-3">
        <CardTitle className="text-sm flex items-center justify-between">
          <span className="flex items-center gap-2">
            <Eye className="h-4 w-4 text-blue-500" />
            Manual Verification Required
          </span>
          <Badge variant="outline" className="text-xs">
            {verification.plan_id}
          </Badge>
        </CardTitle>
        <p className="text-xs text-muted-foreground">
          {verification.plan_name}
        </p>
      </CardHeader>

      <CardContent className="space-y-4">
        {/* Auto QA Status */}
        <div className="flex items-center gap-2 text-xs">
          {verification.auto_qa_passed ? (
            <>
              <CheckCircle2 className="h-4 w-4 text-green-500" />
              <span className="text-green-700">Auto-Claude QA Passed</span>
            </>
          ) : (
            <>
              <AlertCircle className="h-4 w-4 text-yellow-500" />
              <span className="text-yellow-700">Auto-Claude QA Pending</span>
            </>
          )}
        </div>

        {/* Checklist */}
        <div className="space-y-2">
          <p className="text-xs font-medium">Verification Checklist:</p>
          {checklist.map((item) => (
            <div
              key={item.id}
              className="flex items-start gap-2 p-2 rounded bg-background"
            >
              <Checkbox
                id={item.id}
                checked={item.checked}
                onCheckedChange={() => toggleItem(item.id)}
                className="mt-0.5"
              />
              <label
                htmlFor={item.id}
                className={cn(
                  'text-xs cursor-pointer flex-1',
                  item.checked && 'line-through text-muted-foreground'
                )}
              >
                {item.description}
              </label>
            </div>
          ))}
        </div>

        {/* Feedback */}
        <div className="space-y-2">
          <p className="text-xs font-medium">
            Feedback {!allChecked && <span className="text-red-500">*</span>}:
          </p>
          <Textarea
            placeholder="Enter your feedback here... (required for rejection)"
            value={feedback}
            onChange={(e) => setFeedback(e.target.value)}
            className="text-xs min-h-[80px]"
          />
        </div>

        {/* Error */}
        {error && (
          <Alert variant="destructive" className="py-2">
            <AlertDescription className="text-xs">{error}</AlertDescription>
          </Alert>
        )}
      </CardContent>

      <CardFooter className="flex justify-end gap-2 pt-0">
        <Button
          variant="outline"
          size="sm"
          onClick={handleReject}
          disabled={submitting}
          className="text-red-600 hover:text-red-700 hover:bg-red-50"

            <Loader2 className="h-4 w-4 animate-spin mr-1" />
          ) : (
            <ThumbsUp className="h-4 w-4 mr-1" />
          )}
          Approve
        </Button>
      </CardFooter>
    </Card>
  );
}
